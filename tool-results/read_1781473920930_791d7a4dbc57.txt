     3→// ============================================
     4→// 🔸 Smart Retry System for Supabase Queries
     5→// 🔸 Automatically retries on network/fetch errors
     6→// 🔸 Skips retry on data/logic errors (validation, not found, etc.)
     7→// 🔸 Exponential backoff: 500ms → 1000ms → 2000ms
     8→// ============================================
     9→
    10→/**
    11→ * Determines if an error is retryable (network/fetch related).
    12→ * Data/logic errors like "not found", "permission denied", or validation
    13→ * errors are NOT retryable — retrying won't fix them.
    14→ */
    15→function isRetryableError(error: unknown): boolean {
    16→  if (!error) return false;
    17→
    18→  const msg = (error instanceof Error ? error.message : String(error)).toLowerCase();
    19→
    20→  // Network / fetch errors — retryable
    21→  if (
    22→    msg.includes('failed to fetch') ||
    23→    msg.includes('networkerror') ||
    24→    msg.includes('network request failed') ||
    25→    msg.includes('net::err_') ||
    26→    msg.includes('fetch error') ||
    27→    msg.includes('timeout') ||
    28→    msg.includes('aborted') ||
    29→    msg.includes('connection') ||
    30→    msg.includes('econnrefused') ||
    31→    msg.includes('econnreset') ||
    32→    msg.includes('socket hang up') ||
    33→    msg.includes('load failed')
    34→  ) {
    35→    return true;
    36→  }
    37→
    38→  // TypeError in fetch context — usually network-related
    39→  if (error instanceof TypeError && (
    40→    msg.includes('fetch') ||
    41→    msg.includes('network') ||
    42→    msg.includes('load') ||
    43→    msg.includes('failed')
    44→  )) {
    45→    return true;
    46→  }
    47→
    48→  // NOT retryable: table doesn't exist, RLS, validation, business logic
    49→  if (
    50→    msg.includes('could not find') ||
    51→    msg.includes('does not exist') ||
    52→    msg.includes('schema cache') ||
    53→    msg.includes('relation') ||
    54→    msg.includes('policy') ||
    55→    msg.includes('permission') ||
    56→    msg.includes('jwt') ||
    57→    msg.includes('rls') ||
    58→    msg.includes('violates') ||
    59→    msg.includes('duplicate') ||
    60→    msg.includes('unique') ||
    61→    msg.includes('not null') ||
    62→    msg.includes('foreign key') ||
    63→    msg.includes('check constraint')
    64→  ) {
    65→    return false;
    66→  }
    67→
    68→  // For Supabase PostgrestError — check code
    69→  const anyErr = error as Record<string, unknown>;
    70→  const code = String(anyErr?.code || '');
    71→  // Network-related Supabase error codes
    72→  if (code === '08' || code.startsWith('08')) return true; // Connection exception
    73→  if (code === '57' || code.startsWith('57')) return true; // Operator intervention
    74→
    75→  // Default: retry unknown errors — they could be transient network issues
    76→  // (Supabase PostgrestError objects are plain objects, not Error instances,
    77→  // and may have empty messages like {}, so we retry them rather than crash)
    78→  return true;
    79→}
    80→
    81→/**
    82→ * Execute an async function with automatic retry on network errors.
    83→ * Uses exponential backoff: 500ms → 1000ms → 2000ms.
    84→ * Only retries on network/fetch errors — data/logic errors fail immediately.
    85→ *
    86→ * @param fn - The async function to execute
    87→ * @param retries - Maximum number of retries (default: 3)
    88→ * @param delay - Initial delay in ms, doubles on each retry (default: 500)
    89→ * @returns The result of fn(), or null if all retries failed
    90→ */
    91→export async function fetchWithRetry<T>(
    92→  fn: () => Promise<T>,
    93→  retries: number = 3,
    94→  delay: number = 500,
    95→): Promise<T | null> {
    96→  try {
    97→    return await fn();
    98→  } catch (error) {
    99→    // If the error is not retryable, log and return null instead of throwing
   100→    // This prevents app crashes — callers use `return result || []` as fallback
   101→    if (!isRetryableError(error)) {
   102→      console.error('[Supabase] ❌ Non-retryable error (returning null):', error instanceof Error ? error.message : error);
   103→      return null;
   104→    }
   105→
   106→    if (retries <= 0) {
   107→      console.error('[Supabase] ❌ All retries exhausted:', error instanceof Error ? error.message : error);
   108→      return null;
   109→    }
   110→
   111→    const attempt = 3 - retries + 1; // Human-readable attempt number
   112→    console.warn(`[Supabase] ⚠️ Network error (attempt ${attempt}/3), retrying in ${delay}ms...`, error instanceof Error ? error.message : error);
   113→
   114→    await new Promise(res => setTimeout(res, delay));
   115→
   116→    return fetchWithRetry(fn, retries - 1, delay * 2);
   117→  }
   118→}

   332→// ============================================
   333→// Helper: camelCase ↔ snake_case conversion
   334→// ============================================
   335→
   336→function toCamelCaseKey(key: string): string {
   337→  return key.replace(/_([a-z])/g, (_, letter) => letter.toUpperCase());
   338→}
   339→
   340→function toSnakeCaseKey(key: string): string {
   341→  return key.replace(/[A-Z]/g, letter => `_${letter.toLowerCase()}`);
   342→}
   343→
   344→function toCamelCase<T extends Record<string, unknown>>(obj: Record<string, unknown>): T {
   345→  const result: Record<string, unknown> = {};
   346→  for (const [key, value] of Object.entries(obj)) {
   347→    result[toCamelCaseKey(key)] = value;
   348→  }
   349→  return result as T;
   350→}
   351→
   352→function toSnakeCase<T extends Record<string, unknown>>(obj: Record<string, unknown>): T {
   353→  const result: Record<string, unknown> = {};
   354→  for (const [key, value] of Object.entries(obj)) {
   355→    if (value !== undefined) {
   356→      result[toSnakeCaseKey(key)] = value;
   357→    }
   358→  }
   359→  return result as T;
   360→}

   366→function isoToDate(value: string | null | undefined): Date | null | undefined {
   367→  if (value === null) return null;
   368→  if (value === undefined) return undefined;
   369→  return new Date(value);
   370→}
   371→
   372→function dateToIso(value: Date | string | null | undefined): string | null | undefined {
   373→  if (value === null) return null;
   374→  if (value === undefined) return undefined;
   375→  if (value instanceof Date) return value.toISOString();
   376→  return new Date(value).toISOString();
   377→}

   566→// Helper: clean undefined fields for Supabase

   568→
   569→function removeUndefined(obj: Record<string, unknown>): Record<string, unknown> {
   570→  const result: Record<string, unknown> = {};
   571→  for (const [key, value] of Object.entries(obj)) {
   572→    if (value !== undefined) {
   573→      result[key] = value;
   574→    }
   575→  }
   576→  return result;
   577→}

   587→// ============================================
   588→// calculateFinalBalance - COPIED VERBATIM
   589→// ============================================
   590→
   591→function calculateFinalBalance(
   592→  amount: number,
   593→  conversionFactor: number,
   594→  conversionMethod: string,
   595→  feesType: string,
   596→  feesAmount: number,
   597→  feesDirection: string,
   598→  transactionType: 'INCOME' | 'EXPENSE' = 'INCOME'
   599→): number {
   600→  if (!conversionFactor || conversionFactor === 0) {
   601→    return 0;
   602→  }
   603→  
   604→  let finalBalance = amount;
   605→  
   606→  if (conversionMethod === 'MULTIPLY') {
   607→    finalBalance = amount * conversionFactor;
   608→  } else {
   609→    finalBalance = amount / conversionFactor;
   610→  }
   611→  
   612→  let feesValue = 0;
   613→  if (feesAmount && feesAmount > 0) {
   614→    switch (feesType) {
   615→      case 'FIXED':
   616→        feesValue = feesAmount;
   617→        break;
   618→      case 'PERCENTAGE':
   619→        feesValue = (finalBalance * feesAmount) / 100;
   620→        break;
   621→      case 'PER_THOUSAND':
   622→        feesValue = (finalBalance * feesAmount) / 1000;
   623→        break;
   624→    }
   625→    
   626→    const sameDirection = transactionType === feesDirection;
   627→    
   628→    if (sameDirection) {
   629→      finalBalance = finalBalance + feesValue;
   630→    } else {
   631→      finalBalance = finalBalance - feesValue;
   632→    }
   633→  }
   634→  
   635→  return finalBalance;
   636→}

   681→/**
   682→ * Check whether Supabase tables exist by trying a simple SELECT
   683→ * 🔸 Also checks if Supabase client is configured (isSupabaseConfigured)
   684→ * 🔸 Distinguishes between "tables missing" vs "network/RLS error"
   685→ * 🔸 Logs detailed info for debugging on different devices
   686→ */
   687→export async function checkTablesExist(): Promise<boolean> {
   688→  // First check if Supabase is even configured
   689→  if (!isSupabaseConfigured) {
   690→    console.error('[Supabase] ❌ Client not configured — missing env vars');
   691→    console.error('[Supabase] 💡 Check: NEXT_PUBLIC_SUPABASE_URL and NEXT_PUBLIC_SUPABASE_ANON_KEY must be set');
   692→    console.error('[Supabase] 💡 On Vercel: add in Settings → Environment Variables');
   693→    tablesExist = false;
   694→    return false;
   695→  }
   696→
   697→  try {
   698→    console.log('[Supabase] 🔍 Checking if tables exist...');
   699→    const result = await fetchWithRetry(async () => {
   700→      const { data, error, status } = await supabase.from('currencies').select('id').limit(1);
   701→      if (error) throw error;
   702→      return { data, status };
   703→    });
   704→
   705→    if (result === null) {
   706→      // All retries failed — likely network issue, assume tables exist and retry later
   707→      console.warn('[Supabase] ⚠️ checkTablesExist: network error after retries, assuming tables exist');
   708→      tablesExist = true;
   709→      return true;
   710→    }
   711→
   712→    // Success — tables exist
   713→    console.log('[Supabase] ✅ Tables exist, currencies count:', result.data?.length ?? 0);
   714→    tablesExist = true;
   715→    return true;
   716→  } catch (err) {
   717→    // fetchWithRetry re-throws non-retryable errors (e.g., table doesn't exist, RLS)
   718→    const msg = (err instanceof Error ? err.message : String(err)).toLowerCase();
   719→    
   720→    // Table doesn't exist
   721→    if (
   722→      msg.includes('could not find') ||
   723→      msg.includes('does not exist') ||
   724→      msg.includes('schema cache') ||
   725→      msg.includes('relation')
   726→    ) {
   727→      console.error('[Supabase] 💡 Tables do NOT exist. Run the migration SQL in Supabase SQL Editor.');
   728→      tablesExist = false;
   729→      return false;
   730→    }
   731→
   732→    // RLS or auth errors — tables exist but access denied
   733→    if (msg.includes('policy') || msg.includes('permission') || msg.includes('jwt') || msg.includes('rls') || msg.includes('new row violates')) {
   734→      console.error('[Supabase] ⚠️ Tables EXIST but RLS blocks access — FIX: Run fix-rls.sql in Supabase SQL Editor');
   735→      tablesExist = true;
   736→      return true;
   737→    }
   738→
   739→    // Other errors — don't block the app
   740→    console.error('[Supabase] ❌ checkTablesExist exception:', err);
   741→    tablesExist = true;
   742→    return true;
   743→  }
   744→}

   776→      tablesExist = false;

  1110→/**
  1111→ * Recalculate vault balance from opening_balance + all vault-affecting operations after opening_balance_date
  1112→ * Formula: balance = opening_balance + sum(post-date operations)
  1113→ *
  1114→ * Operations that affect vault:
  1115→ * 1. TRANSACTIONS: Complete, cash, not archived
  1116→ * 2. DEBTS: Cash mode debts
  1117→ * 3. DEBT PAYMENTS: Cash mode payments
  1118→ * 4. CURRENCY EXCHANGES: Not deleted
  1119→ */
  1120→export async function recalculateVaultBalance(currencyId: string): Promise<Vault | null> {
  1121→  await initializeDatabase();
  1122→
  1123→  // Get the vault
  1124→  const { data: vaultRows } = await supabase.from('vaults').select('*').eq('currency_id', currencyId);
  1125→  if (!vaultRows || vaultRows.length === 0) return null;
  1126→  const vault = rowToVault(vaultRows[0]);
  1127→
  1128→  const openingBalance = vault.openingBalance || 0;
  1129→  const openingDate = vault.openingBalanceDate;
  1130→
  1131→  let delta = 0;
  1132→
  1133→  // Build date filter - only operations AFTER opening_balance_date
  1134→  const dateFilter = openingDate ? openingDate.toISOString() : null;
  1135→
  1136→  // 1. TRANSACTIONS: Complete, cash (overflow transactions use currencyId directly)
  1137→  let transactionQuery = supabase.from('transactions').select('*')
  1138→    .eq('is_complete', true)
  1139→    .eq('payment_type', 'CASH');
  1140→
  1141→  if (dateFilter) {
  1142→    transactionQuery = transactionQuery.gt('date', dateFilter);
  1143→  }
  1144→
  1145→  const { data: transactions } = await transactionQuery;
  1146→  if (transactions) {
  1147→    for (const tx of transactions) {
  1148→      const txObj = rowToTransaction(tx);
  1149→      // Determine which vault this transaction affects
  1150→      const vaultCurrencyId = txObj.isOverflowTransaction ? txObj.currencyId : (txObj.baseCurrencyId || txObj.currencyId);
  1151→
  1152→      if (vaultCurrencyId === currencyId) {
  1153→        // INCOME: vault decreases (money goes out)
  1154→        // EXPENSE: vault increases (money comes in)
  1155→        if (txObj.type === 'INCOME') {
  1156→          delta -= txObj.amount;
  1157→        } else {
  1158→          delta += txObj.amount;
  1159→        }
  1160→      }
  1161→    }
  1162→  }
  1163→
  1164→  // 2. DEBTS: Cash mode debts affect vault
  1165→  let debtQuery = supabase.from('debts').select('*')
  1166→    .eq('debt_mode', 'CASH');
  1167→
  1168→  if (dateFilter) {
  1169→    debtQuery = debtQuery.gt('date', dateFilter);
  1170→  }
  1171→
  1172→  const { data: debts } = await debtQuery;
  1173→  if (debts) {
  1174→    for (const d of debts) {
  1175→      const debtObj = rowToDebt(d);
  1176→      if (debtObj.currencyId === currencyId) {
  1177→        // PAYABLE (علينا): vault increases (we receive money)
  1178→        // RECEIVABLE (لنا): vault decreases (we give money out)
  1179→        if (debtObj.debtType === 'PAYABLE') {
  1180→          delta += debtObj.amount;
  1181→        } else {
  1182→          delta -= debtObj.amount;
  1183→        }
  1184→      }
  1185→    }
  1186→  }
  1187→
  1188→  // 3. DEBT PAYMENTS: Cash mode
  1189→  let paymentQuery = supabase.from('debt_payments').select('*')
  1190→    .eq('payment_mode', 'CASH');
  1191→
  1192→  if (dateFilter) {
  1193→    paymentQuery = paymentQuery.gt('date', dateFilter);
  1194→  }
  1195→
  1196→  const { data: payments } = await paymentQuery;
  1197→  if (payments) {
  1198→    for (const p of payments) {
  1199→      const paymentObj = rowToDebtPayment(p);
  1200→      if (paymentObj.currencyId === currencyId) {
  1201→        // Debt payment direction logic:
  1202→        // RECEIVABLE (لنا): vault decreases (money goes out to pay toward our receivable)
  1203→        // PAYABLE (علينا): vault increases (money comes in to pay toward our payable)
  1204→        const direction = paymentObj.paymentDirection || 'RECEIVABLE';
  1205→        if (direction === 'PAYABLE') {
  1206→          delta += paymentObj.amount;
  1207→        } else {
  1208→          delta -= paymentObj.amount;
  1209→        }
  1210→      }
  1211→    }
  1212→  }
  1213→
  1214→  // 4. CURRENCY EXCHANGES: Not deleted
  1215→  let exchangeQuery = supabase.from('currency_exchanges').select('*')
  1216→    .eq('is_deleted', false);
  1217→
  1218→  if (dateFilter) {
  1219→    exchangeQuery = exchangeQuery.gt('date', dateFilter);
  1220→  }
  1221→
  1222→  const { data: exchanges } = await exchangeQuery;
  1223→  if (exchanges) {
  1224→    for (const ex of exchanges) {
  1225→      const exchangeObj = rowToCurrencyExchange(ex);
  1226→      // Outgoing: vault decreases
  1227→      if (exchangeObj.outgoingCurrencyId === currencyId) {
  1228→        delta -= exchangeObj.outgoingAmount;
  1229→      }
  1230→      // Incoming: vault increases
  1231→      if (exchangeObj.incomingCurrencyId === currencyId) {
  1232→        delta += exchangeObj.incomingAmount;
  1233→      }
  1234→    }
  1235→  }
  1236→
  1237→  // Compute new balance
  1238→  const newBalance = openingBalance + delta;
  1239→
  1240→  // Update vault
  1241→  const { error } = await supabase.from('vaults').update({
  1242→    balance: newBalance,
  1243→    updated_at: new Date().toISOString(),
  1244→  }).eq('id', vault.id);
  1245→
  1246→  if (error) throw new Error(error.message);
  1247→
  1248→  const { data: updatedRow } = await supabase.from('vaults').select('*').eq('id', vault.id).single();
  1249→  return updatedRow ? rowToVault(updatedRow) : null;
  1250→}
