    91→// ============================================
    92→// Helper: Retry with exponential backoff
    93→// ============================================
    94→async function withRetry<T>(fn: () => Promise<T>, retries = MAX_RETRIES, delay = RETRY_DELAY_MS): Promise<T> {
    95→  for (let attempt = 1; attempt <= retries; attempt++) {
    96→    try {
    97→      return await fn();
    98→    } catch (error) {
    99→      if (attempt === retries) throw error;
   100→      console.warn(`Attempt ${attempt}/${retries} failed, retrying after ${delay}ms...`, error);
   101→      await new Promise(resolve => setTimeout(resolve, delay * attempt));
   102→    }
   103→  }
   104→  throw new Error('All retry attempts failed');
   105→}

   153→  const refreshData = useCallback(async (skipEvent = false, forceRefresh = false) => {
   154→    // If a refresh is in progress and this is not forced, skip
   155→    if (isRefreshingRef.current && !forceRefresh) return;

   160→        await new Promise(r => setTimeout(r, 100));
   161→        if (!isRefreshingRef.current) break;
   162→      }
   163→      if (isRefreshingRef.current) return; // timeout, skip
   164→    }
   165→
   166→    // Debounce: prevent rapid consecutive refreshes (unless forced)
   167→    if (!forceRefresh) {
   168→      const debounceNow = Date.now();
   169→      if (debounceNow - lastRefreshTimeRef.current < REFRESH_DEBOUNCE_MS) return;

   173→    lastRefreshTimeRef.current = Date.now();

   184→        getTotalBalanceInUSD(),

   191→      setCurrencies(activeCur);
   192→      setVaults(vaultsData);
   193→      setAccounts(accountsData);
   194→      setTransactions(txData);
   195→      setDebts(debtData);
   196→      setDebtPayments(debtPaymentsData);
   197→      setTotalBalanceUSD(totalUSD);
   198→      // debtRemaining is now computed client-side via useMemo (computedDebtRemaining)
   199→      setCurrencyExchanges(exchangeData);
   200→      setInitError(null);
   201→
   202→      // Dispatch event for other components (same as useLocalData)
   203→      if (!skipEvent) {
   204→        window.dispatchEvent(new CustomEvent('local-data-refreshed'));
   205→      }
   206→    } catch (error) {

   208→      setInitError('حدث خطأ في تحميل البيانات');
   209→    } finally {
   210→      isRefreshingRef.current = false;
   211→    }
   212→  }, []);

   297→  // ============================================

   302→
   303→  const debouncedRefresh = useCallback((table: string, refreshFn: () => Promise<void>) => {
   304→    const existing = debouncedRefreshMap.current.get(table);
   305→    if (existing) clearTimeout(existing);
   306→
   307→    const timer = setTimeout(() => {
   308→      debouncedRefreshMap.current.delete(table);
   309→      refreshFn();
   310→    }, REFRESH_DEBOUNCE_MS);
   311→
   312→    debouncedRefreshMap.current.set(table, timer);
   313→  }, []);
   314→

   316→  // Refs for Realtime callback functions
   317→  // Using refs so the Realtime effect only runs ONCE and always
   318→  // calls the latest callback versions without re-subscribing
   319→  // ============================================
   320→  const refreshCurrenciesRef = useRef(refreshCurrencies);
   321→  const refreshVaultsRef = useRef(refreshVaults);
   322→  const refreshAccountsRef = useRef(refreshAccounts);
   323→  const refreshTransactionsRef = useRef(refreshTransactions);
   324→  const refreshDebtsRef = useRef(refreshDebts);
   325→  const refreshDebtPaymentsRef = useRef(refreshDebtPayments);

   328→  const refreshDataRef = useRef(refreshData);

   334→  refreshTransactionsRef.current = refreshTransactions;
   335→  refreshDebtsRef.current = refreshDebts;
   336→  refreshDebtPaymentsRef.current = refreshDebtPayments;
   337→  refreshCurrencyExchangesRef.current = refreshCurrencyExchanges;
   338→  debouncedRefreshRef.current = debouncedRefresh;
   339→  refreshDataRef.current = refreshData;
   340→
   341→  // ============================================
   342→  // Realtime Subscriptions
   343→  // 🔸 Effect runs ONCE on mount (empty dependency array)
   344→  // 🔸 Uses refs to call latest callback versions
   345→  // 🔸 Uses unique channel name to prevent "already subscribed" conflicts
   346→  // 🔸 Skips subscription if Supabase tables don't exist
   347→  // ============================================
   348→  useEffect(() => {
   349→    mountedRef.current = true;
   350→
   351→    // Async setup: check tables exist before subscribing
   352→    const setupRealtime = async () => {
   353→      // Check if tables exist before attempting Realtime subscription
   354→      const exist = await checkTablesExist();
   355→      tablesExistRef.current = exist;
   356→
   357→      if (!exist) {
   358→        console.warn('Supabase Realtime: tables do not exist, skipping subscription');
   359→        return;
   360→      }
   361→
   362→      // Use a unique channel name to prevent conflicts with stale channels
   363→      // from React StrictMode double-mounting or hot reloads
   364→      const channelId = `schema-db-changes-${Date.now()}`;
   365→
   366→      const channel = supabase
   367→        .channel(channelId)
   368→        .on('postgres_changes', { event: '*', schema: 'public', table: 'currencies' }, () => {
   369→          debouncedRefreshRef.current('currencies', () => refreshCurrenciesRef.current());
   370→        })
   371→        .on('postgres_changes', { event: '*', schema: 'public', table: 'vaults' }, () => {
   372→          // Vaults depend on currencies for total balance calculation
   373→          debouncedRefreshRef.current('vaults', async () => {
   374→            await refreshVaultsRef.current();
   375→            // Also refresh total balance since it depends on vaults
   376→            try {
   377→              const totalUSD = await getTotalBalanceInUSD();
   378→              if (mountedRef.current) setTotalBalanceUSD(totalUSD);
   379→            } catch (e) {
   380→              console.error('Error refreshing total balance:', e);
   381→            }
   382→          });
   383→        })
   384→        .on('postgres_changes', { event: '*', schema: 'public', table: 'accounts' }, () => {
   385→          debouncedRefreshRef.current('accounts', () => refreshAccountsRef.current());
   386→        })
   387→        .on('postgres_changes', { event: '*', schema: 'public', table: 'transactions' }, () => {
   388→          debouncedRefreshRef.current('transactions', () => refreshTransactionsRef.current());
   389→        })
   390→        .on('postgres_changes', { event: '*', schema: 'public', table: 'debts' }, () => {
   391→          debouncedRefreshRef.current('debts', () => refreshDebtsRef.current());
   392→        })
   393→        .on('postgres_changes', { event: '*', schema: 'public', table: 'debt_payments' }, () => {
   394→          debouncedRefreshRef.current('debt_payments', () => refreshDebtPaymentsRef.current());
   395→        })
   396→        .on('postgres_changes', { event: '*', schema: 'public', table: 'currency_exchanges' }, () => {
   397→          debouncedRefreshRef.current('currency_exchanges', () => refreshCurrencyExchangesRef.current());
   398→        })
   399→        .subscribe((status) => {
   400→          if (status === 'SUBSCRIBED') {
   401→            realtimeConnectedRef.current = true;
   402→            console.log('Supabase Realtime: ✅ subscribed successfully');
   403→            // Keep periodic polling even with Realtime — ensures cross-device sync reliability
   404→            // (Realtime only works while tab is open; polling catches up after tab reactivation)
   405→            if (!fallbackPollRef.current) {
   406→              fallbackPollRef.current = setInterval(() => {
   407→                refreshDataRef.current(true);
   408→              }, FALLBACK_POLL_INTERVAL_MS);
   409→            }
   410→          } else if (status === 'CHANNEL_ERROR' || status === 'TIMED_OUT') {
   411→            realtimeConnectedRef.current = false;
   412→            console.warn('Supabase Realtime: ❌ subscription error, using polling only');
   413→            // Start fallback polling if not already running
   414→            if (!fallbackPollRef.current) {
   415→              fallbackPollRef.current = setInterval(() => {
   416→                refreshDataRef.current(true);
   417→              }, FALLBACK_POLL_INTERVAL_MS);
   418→            }
   419→          }
   420→        });
   421→
   422→      channelRef.current = channel;
   423→
   424→      // 🔸 ALWAYS start periodic polling for reliable cross-device sync
   425→      // Even when Realtime works, polling ensures data consistency
   426→      if (!fallbackPollRef.current) {
   427→        fallbackPollRef.current = setInterval(() => {
   428→          refreshDataRef.current(true);
   429→        }, FALLBACK_POLL_INTERVAL_MS);
   430→      }
   431→    };
   432→
   433→    setupRealtime();
   434→
   435→    return () => {
   436→      mountedRef.current = false;
   437→      // Clean up debounced timers
   438→      debouncedRefreshMap.current.forEach(timer => clearTimeout(timer));
   439→      debouncedRefreshMap.current.clear();
   440→      // Unsubscribe first, then remove channel
   441→      if (channelRef.current) {
   442→        channelRef.current.unsubscribe();
   443→        supabase.removeChannel(channelRef.current);
   444→        channelRef.current = null;
   445→      }
   446→      // Clean up fallback polling
   447→      if (fallbackPollRef.current) {
   448→        clearInterval(fallbackPollRef.current);
   449→        fallbackPollRef.current = null;
   450→      }
   451→    };
   452→  }, []); // Intentionally empty — uses refs for callbacks
   453→
   454→  // ============================================
   455→  // Initialization with retry + timeout
   456→  // 🔸 Checks if Supabase is configured first
   457→  // 🔸 More resilient error handling for cross-device support
   458→  // ============================================
   459→  useEffect(() => {
   460→    let cancelled = false;
   461→    let timeoutId: ReturnType<typeof setTimeout>;
   462→
   463→    const init = async () => {
   464→      try {
   465→        setIsLoading(true);
   466→        setInitError(null);
   467→
   468→        // 🔸 First: Check if Supabase client is configured
   469→        if (!isSupabaseConfigured) {
   470→          console.error('[Supabase] ❌ Client not configured — env vars missing');
   471→          setTablesMissing(true);
   472→          setInitError('لم يتم تكوين Supabase. تأكد من إضافة متغيرات البيئة NEXT_PUBLIC_SUPABASE_URL و NEXT_PUBLIC_SUPABASE_ANON_KEY');
   473→          setIsInitialized(true);
   474→          setIsLoading(false);
   475→          return;
   476→        }
   477→
   478→        console.log('[Supabase] 🔄 Initializing database...');
   479→
   480→        await withRetry(async () => {
   481→          if (cancelled) return;
   482→          await initializeDatabase();
   483→        });
   484→
   485→        if (cancelled) return;
   486→
   487→        // After initializeDatabase(), check if tables are missing
   488→        const tablesDoExist = await checkTablesExist();
   489→        tablesExistRef.current = tablesDoExist;
   490→
   491→        if (!tablesDoExist) {
   492→          console.warn('[Supabase] ⚠️ Tables missing — showing setup screen');
   493→          setTablesMissing(true);
   494→          setInitError('لم يتم العثور على جداول قاعدة البيانات. يرجى إعداد قاعدة البيانات أولاً.');
   495→          setIsInitialized(true);
   496→          setIsLoading(false);
   497→          return;
   498→        }
   499→
   500→        console.log('[Supabase] ✅ Tables exist, loading data...');
   501→        setTablesMissing(false);
   502→        await refreshData(true); // skipEvent=true to prevent event during init
   503→
   504→        if (!cancelled) {
   505→          setIsInitialized(true);
   506→          console.log('[Supabase] ✅ Initialization complete!');
   507→        }
   508→      } catch (error) {
   509→        console.error('[Supabase] ❌ Initialization failed:', error);
   510→        if (!cancelled) {
   511→          setInitError('فشل في تهيئة قاعدة البيانات. يرجى إعادة تحميل الصفحة.');
   512→          setIsInitialized(true); // Allow entry to prevent white screen
   513→        }
   514→      } finally {
   515→        if (!cancelled) {
   516→          setIsLoading(false);
   517→        }
   518→      }
   519→    };
   520→
   521→    // Timeout: prevent infinite waiting
   522→    timeoutId = setTimeout(() => {
   523→      if (!cancelled && !isInitialized) {
   524→        console.warn('[Supabase] ⏱️ Initialization timeout');
   525→        setIsInitialized(true);
   526→        setIsLoading(false);
   527→        setInitError('استغرقت التهيئة وقتًا طويلاً. قد لا تكون جميع البيانات محملة.');
   528→      }
   529→    }, INIT_TIMEOUT_MS);
   530→
   531→    init();
   532→
   533→    return () => {
   534→      cancelled = true;
   535→      clearTimeout(timeoutId);
   536→    };
   537→  }, [refreshData]);
   538→
   539→  // ============================================
   540→  // Listen for data refresh events (from other components)
   541→  // ============================================
   542→  useEffect(() => {
   543→    let cancelled = false;
   544→
   545→    const handleDataRefresh = async () => {
   546→      if (cancelled || isRefreshingRef.current) return;
   547→      await refreshData(true);
   548→    };
   549→
   550→    window.addEventListener('local-data-refreshed', handleDataRefresh);
   551→    window.addEventListener('app-data-refreshed', handleDataRefresh);
   552→
   553→    return () => {
   554→      cancelled = true;
   555→      window.removeEventListener('local-data-refreshed', handleDataRefresh);
   556→      window.removeEventListener('app-data-refreshed', handleDataRefresh);
   557→    };
   558→  }, [refreshData]);
   559→

   561→  // Network restore: reload data when connection is back

   577→    window.addEventListener('app-network-restored', handleNetworkRestored);
   578→    window.addEventListener('online', handleNetworkRestored);
   579→
   580→    return () => {
   581→      cancelled = true;
   582→      window.removeEventListener('app-network-restored', handleNetworkRestored);
   583→      window.removeEventListener('online', handleNetworkRestored);

   659→    try {
   660→      await initializeDatabase();
   661→
   662→      // Check if tables now exist (after setup)
   663→      const tablesDoExist = await checkTablesExist();

   666→      if (!tablesDoExist) {
   667→        // Still no tables — keep showing setup screen

   669→        setInitError('لم يتم العثور على جداول قاعدة البيانات. يرجى إعداد قاعدة البيانات أولاً.');
   670→        setIsInitialized(true);
   671→        setIsLoading(false);
   672→        return;
   673→      }
   674→
   675→      // Tables exist now! Clear the missing flag and reload data
   676→      setTablesMissing(false);
   677→      await refreshData(true);
   678→      setIsInitialized(true);
   679→
   680→      // Set up Realtime subscription if we don't have one yet
   681→      if (!channelRef.current) {
   682→        const channelId = `schema-db-changes-${Date.now()}`;
   683→        const channel = supabase
   684→          .channel(channelId)
   685→          .on('postgres_changes', { event: '*', schema: 'public', table: 'currencies' }, () => {
   686→            debouncedRefreshRef.current('currencies', () => refreshCurrenciesRef.current());

   694→              } catch (e) {
   695→                console.error('Error refreshing total balance:', e);
   696→              }
   697→            });
   698→          })
   699→          .on('postgres_changes', { event: '*', schema: 'public', table: 'accounts' }, () => {
   700→            debouncedRefreshRef.current('accounts', () => refreshAccountsRef.current());
   701→          })
   702→          .on('postgres_changes', { event: '*', schema: 'public', table: 'transactions' }, () => {
   703→            debouncedRefreshRef.current('transactions', () => refreshTransactionsRef.current());
   704→          })
   705→          .on('postgres_changes', { event: '*', schema: 'public', table: 'debts' }, () => {
   706→            debouncedRefreshRef.current('debts', () => refreshDebtsRef.current());
   707→          })
   708→          .on('postgres_changes', { event: '*', schema: 'public', table: 'debt_payments' }, () => {
   709→            debouncedRefreshRef.current('debt_payments', () => refreshDebtPaymentsRef.current());
   710→          })
   711→          .on('postgres_changes', { event: '*', schema: 'public', table: 'currency_exchanges' }, () => {
   712→            debouncedRefreshRef.current('currency_exchanges', () => refreshCurrencyExchangesRef.current());
   713→          })

   716→              realtimeConnectedRef.current = true;
   717→              console.log('Supabase Realtime: subscribed after retry');
   718→            }
   719→          });
   720→        channelRef.current = channel;
   721→      }

   724→      const msg = error instanceof Error ? error.message : 'فشلت إعادة المحاولة';
   725→      setInitError(msg);
   726→    } finally {
   727→      setIsLoading(false);
   728→    }
