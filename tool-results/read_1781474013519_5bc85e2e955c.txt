   288→function calculateFinalBalance(
   289→  amount: number,
   290→  conversionFactor: number,
   291→  conversionMethod: string,
   292→  feesType: string,
   293→  feesAmount: number,
   294→  feesDirection: string,
   295→  transactionType: 'INCOME' | 'EXPENSE' = 'INCOME'
   296→): number {
   297→  // 🔸 حماية: إذا كان معامل التحويل = 0، الرصيد النهائي = 0
   298→  // (لتجنب القسمة على صفر في وضع DIVIDE)
   299→  if (!conversionFactor || conversionFactor === 0) {
   300→    return 0;
   301→  }
   302→  
   303→  let finalBalance = amount;
   304→  
   305→  if (conversionMethod === 'MULTIPLY') {
   306→    finalBalance = amount * conversionFactor;
   307→  } else {
   308→    finalBalance = amount / conversionFactor;
   309→  }
   310→  
   311→  let feesValue = 0;
   312→  if (feesAmount && feesAmount > 0) {
   313→    switch (feesType) {
   314→      case 'FIXED':
   315→        feesValue = feesAmount;
   316→        break;
   317→      case 'PERCENTAGE':
   318→        feesValue = (finalBalance * feesAmount) / 100;
   319→        break;
   320→      case 'PER_THOUSAND':
   321→        feesValue = (finalBalance * feesAmount) / 1000;
   322→        break;
   323→    }
   324→    
   325→    // منطق الأجور:
   326→    // إذا تطابق اتجاه الحركة مع اتجاه الأجور = زيادة في المبلغ النهائي
   327→    // إذا اختلف اتجاه الحركة عن اتجاه الأجور = خصم من المبلغ النهائي
   328→    const sameDirection = transactionType === feesDirection;
   329→    
   330→    if (sameDirection) {
   331→      // نفس الاتجاه = زيادة
   332→      finalBalance = finalBalance + feesValue;
   333→    } else {
   334→      // اتجاه مختلف = خصم
   335→      finalBalance = finalBalance - feesValue;
   336→    }
   337→  }
   338→  
   339→  return finalBalance;
   340→}

   373→export async function initializeDatabase(): Promise<void> {
   374→  if (isDbInitialized) return;
   375→  
   376→  const count = await db.table('currencies').count();
   377→  const usdCurrency = await db.table<Currency>('currencies').get('cur_usd');
   378→  
   379→  // إذا تم حذف العملات، أعد إضافتها
   380→  if (count === 0 || !usdCurrency) {
   381→    const now = new Date();
   382→    
   383→    // حذف أي بيانات قديمة
   384→    await db.table('currencies').clear();
   385→    
   386→    // إضافة العملات الافتراضية
   387→    await db.table('currencies').bulkAdd(
   388→      defaultCurrencies.map(c => ({ ...c, createdAt: now, updatedAt: now }))
   389→    );
   390→    
   391→    // التحقق من وجود صندوق الدولار
   392→    const usdVault = await db.table<Vault>('vaults').where('currencyId').equals('cur_usd').first();
   393→    if (!usdVault) {
   394→      await db.table('vaults').add({
   395→        id: 'vault_cur_usd',
   396→        currencyId: 'cur_usd',
   397→        balance: 0,
   398→        openingBalance: 0,
   399→        createdAt: now,
   400→        updatedAt: now,
   401→      });
   402→    }
   403→    
   404→    // التحقق من وجود حساب افتراضي
   405→    const accountsCount = await db.table('accounts').count();
   406→    if (accountsCount === 0) {
   407→      await db.table('accounts').add({
   408→        id: 'acc_sample',
   409→        name: 'حساب رئيسي',
   410→        type: 'PRIVATE',
   411→        description: 'الحساب الافتراضي',
   412→        isActive: true,
   413→        createdAt: now,
   414→        updatedAt: now,
   415→      });
   416→    }
   417→  }
   418→  
   419→  // ============================================
   420→  // 🔹 إصلاح سير العملة السورية (SYP)
   421→  // تصحيح conversionMethod من DIVIDE إلى MULTIPLY
   422→  // لأن exchangeRate = 0.00004 يمثل "1 ليرة = 0.00004 دولار"
   423→  // والعملية الصحيحة هي: amount × 0.00004 = usdValue (MULTIPLY)
   424→  // وليس: amount ÷ 0.00004 = نتيجة خاطئة
   425→  // ============================================
   426→  try {
   427→    const sypCurrency = await db.table<Currency>('currencies').get('cur_syp');
   428→    if (sypCurrency && sypCurrency.conversionMethod === 'DIVIDE' && sypCurrency.exchangeRate < 1) {
   429→      await db.table('currencies').update('cur_syp', { 
   430→        conversionMethod: 'MULTIPLY',
   431→        updatedAt: new Date()
   432→      });
   433→    }
   434→  } catch (e) {
   435→    // تجاهل الأخطاء في عملية الإصلاح
   436→    console.warn('SYP migration fix skipped:', e);
   437→  }
   438→  
   439→  isDbInitialized = true;
   440→}

   703→export async function addTransaction(data: {
   704→  accountId: string;
   705→  currencyId: string;
   706→  baseCurrencyId?: string | null;
   707→  type: 'INCOME' | 'EXPENSE';
   708→  paymentType: 'CASH' | 'DEFERRED';
   709→  amount: number;
   710→  conversionFactor: number;
   711→  conversionMethod: 'MULTIPLY' | 'DIVIDE';
   712→  feesType: 'FIXED' | 'PERCENTAGE' | 'PER_THOUSAND';
   713→  feesDirection: 'INCOME' | 'EXPENSE';
   714→  feesAmount: number;
   715→  description?: string;
   716→  date: string;
   717→  isOverflowTransaction?: boolean;
   718→  relatedPaymentId?: string | null;
   719→  isComplete?: boolean;
   720→}): Promise<Transaction> {

   725→  // ملاحظة: نستخدم ?? بدلاً من || لأن conversionFactor يمكن أن يكون 0 (قيمة صحيحة)

   727→  const finalBalance = calculateFinalBalance(
   728→    data.amount || 0,

   739→  const isDataComplete = !!(effectiveFactor && effectiveFactor !== 0

   742→  const isComplete = data.isComplete !== undefined ? data.isComplete : isDataComplete;

   750→      throw new Error(`الرصيد غير كافٍ في صندوق ${currency?.name || ''}`);

   760→    paymentType: data.paymentType, 
   761→    amount: data.amount, 
   762→    conversionFactor: effectiveFactor, 
   763→    conversionMethod: data.conversionMethod || 'MULTIPLY', 
   764→    feesType: data.feesType || 'FIXED', 
   765→    feesDirection: data.feesDirection || 'INCOME', 
   766→    feesAmount: data.feesAmount || 0, 
   767→    finalBalance,

   770→    isOverflowTransaction: data.isOverflowTransaction || false,

   774→    updatedAt: now

   786→    }
   787→  }

   792→export async function updateTransaction(id: string, data: Partial<Transaction>): Promise<Transaction> {
   793→  await initializeDatabase();
   794→  const old = await db.table<Transaction>('transactions').get(id);
   795→  if (!old) throw new Error('الحركة غير موجودة');
   796→  
   797→  const now = new Date();
   798→  const oldIsComplete = old.isComplete !== undefined ? old.isComplete : true; // التوافق مع البيانات القديمة
   799→  
   800→  // 🔸 استخدام القيم الجديدة إذا وُجدت، وإلا القيم القديمة
   801→  // ملاحظة: نستخدم ?? بدلاً من || لأن القيمة 0 قيمة صحيحة يجب احترامها
   802→  const effectiveAmount = data.amount ?? old.amount;
   803→  const effectiveConversionFactor = data.conversionFactor ?? old.conversionFactor;
   804→  const effectiveConversionMethod = data.conversionMethod ?? old.conversionMethod;
   805→  const effectiveFeesType = data.feesType ?? old.feesType;
   806→  const effectiveFeesAmount = data.feesAmount ?? old.feesAmount;
   807→  const effectiveFeesDirection = data.feesDirection ?? old.feesDirection;
   808→  const effectiveType = data.type ?? old.type;
   809→  const effectivePaymentType = data.paymentType ?? old.paymentType;
   810→  
   811→  // 🔸 عكس التأثير على الصندوق فقط إذا كانت الحركة القديمة مكتملة ونقدية
   812→  if (oldIsComplete && old.paymentType === 'CASH') {
   813→    const oldVaultCurrencyId = old.baseCurrencyId || old.currencyId;
   814→    const oldVault = await db.table<Vault>('vaults').where('currencyId').equals(oldVaultCurrencyId).first();
   815→    if (oldVault) {
   816→      const reversed = old.type === 'INCOME' ? oldVault.balance + old.amount : oldVault.balance - old.amount;
   817→      await db.table('vaults').update(oldVault.id, { balance: reversed, updatedAt: now });
   818→    }
   819→  }
   820→  
   821→  const finalBalance = calculateFinalBalance(
   822→    effectiveAmount,
   823→    effectiveConversionFactor, 
   824→    effectiveConversionMethod, 
   825→    effectiveFeesType, 
   826→    effectiveFeesAmount, 
   827→    effectiveFeesDirection,
   828→    effectiveType
   829→  );
   830→  
   831→  // 🔸 تحديد حالة الاكتمال تلقائيًا:
   832→  // إذا كانت جميع البيانات مكتملة (مبلغ > 0 + معامل تحويل > 0 + رصيد نهائي > 0) → مكتملة
   833→  // إذا نقص أي حقل → غير مكتملة تلقائيًا
   834→  const isDataComplete = effectiveAmount > 0 && effectiveConversionFactor > 0 && finalBalance > 0;
   835→  const newIsComplete = data.isComplete !== undefined 
   836→    ? (isDataComplete && data.isComplete)  // إذا مرر isComplete يدويًا، نحترمه فقط إذا البيانات مكتملة
   837→    : isDataComplete;  // تلقائي حسب البيانات
   838→  
   839→  // 🔸 تنظيف البيانات - إزالة الحقول غير المعرفة (undefined)
   840→  const cleanData: Record<string, unknown> = {};
   841→  for (const [key, value] of Object.entries(data)) {
   842→    if (value !== undefined) {
   843→      cleanData[key] = value;
   844→    }
   845→  }
   846→  
   847→  await db.table('transactions').update(id, { ...cleanData, finalBalance, isComplete: newIsComplete, date: data.date ? new Date(data.date) : old.date, updatedAt: now });
   848→  
   849→  // 🔸 تطبيق التأثير على الصندوق فقط إذا كانت الحركة الجديدة مكتملة ونقدية
   850→  if (newIsComplete && effectivePaymentType === 'CASH') {
   851→    const vaultCurrencyId = (data.baseCurrencyId ?? data.currencyId) ?? (old.baseCurrencyId || old.currencyId);
   852→    const vault = await db.table<Vault>('vaults').where('currencyId').equals(vaultCurrencyId).first();
   853→    if (vault) {
   854→      const newBalance = effectiveType === 'INCOME' ? vault.balance - effectiveAmount : vault.balance + effectiveAmount;
   855→      await db.table('vaults').update(vault.id, { balance: newBalance, updatedAt: now });
   856→    }
   857→  }
   858→  
   859→  return (await db.table<Transaction>('transactions').get(id))!;
   860→}

   862→export async function deleteTransaction(id: string): Promise<void> {
   863→  await initializeDatabase();
   864→  const transaction = await db.table<Transaction>('transactions').get(id);
   865→  if (!transaction) throw new Error('الحركة غير موجودة');
   866→  
   867→  // ============================================
   868→  // 🔹 عكس التأثير على الصندوق للحركات النقدية
   869→  // 🔸 للحركات العادية: نستخدم baseCurrencyId || currencyId
   870→  // 🔸 لحركات الفائض: نستخدم currencyId مباشرة
   871→  // 🔸 INCOME (لنا) = كان خصم → نزيد
   872→  // 🔸 EXPENSE (علينا) = كانت إضافة → نخصم
   873→  // ============================================
   874→  // 🔸 عكس التأثير على الصندوق فقط للحركات المكتملة والنقدية
   875→  if ((transaction.isComplete !== false) && transaction.paymentType === 'CASH') {
   876→    // تحديد عملة الصندوق
   877→    // لحركات الفائض: نستخدم currencyId مباشرة لأن baseCurrencyId يكون null
   878→    // للحركات العادية: نستخدم baseCurrencyId إذا وجد، وإلا currencyId
   879→    const vaultCurrencyId = transaction.isOverflowTransaction 
   880→      ? transaction.currencyId 
   881→      : (transaction.baseCurrencyId || transaction.currencyId);
   882→    
   883→    const vault = await db.table<Vault>('vaults').where('currencyId').equals(vaultCurrencyId).first();
   884→    
   885→    if (vault) {
   886→      let newBalance = vault.balance;
   887→      
   888→      // ============================================
   889→      // 🔹 منطق عكس التأثير:
   890→      // 🔸 INCOME (لنا): كان خصم من الصندوق → نزيده
   891→      // 🔸 EXPENSE (علينا): كانت إضافة للصندوق → ننقصه
   892→      // ============================================
   893→      if (transaction.type === 'INCOME') {
   894→        // INCOME = كان خصم من الصندوق → نزيده
   895→        newBalance = vault.balance + transaction.amount;
   896→      } else {
   897→        // EXPENSE = كانت إضافة للصندوق → ننقصه
   898→        newBalance = vault.balance - transaction.amount;
   899→      }
   900→      
   901→      await db.table('vaults').update(vault.id, { balance: newBalance, updatedAt: new Date() });
   902→    }
   903→  }
   904→  
   905→  // ============================================
