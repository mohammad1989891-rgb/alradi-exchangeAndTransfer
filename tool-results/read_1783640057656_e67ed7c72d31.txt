    35→// Helper function to format number with thousand separator for input
    36→function formatInputNumber(num: number | string): string {
    37→  if (num === '' || num === null || num === undefined) return '';
    38→  const str = String(num).replace(/,/g, '');
    39→  if (isNaN(parseFloat(str))) return '';
    40→  
    41→  const parts = str.split('.');
    42→  parts[0] = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, ',');
    43→  return parts.join('.');
    44→}
    45→
    46→// Helper function to parse formatted number
    47→function parseFormattedNumber(str: string): number {
    48→  if (!str) return 0;
    49→  return parseFloat(str.replace(/,/g, '')) || 0;
    50→}
