     3→// ============================================
     4→// 🔸 Smart Retry System for Supabase Queries
     5→// 🔸 Automatically retries on network/fetch errors
     6→// 🔸 Skips retry on data/logic errors (validation, not found, etc.)
     7→// 🔸 Exponential backoff: 500ms → 1000ms → 2000ms
     8→// ============================================
     9→
    10→/**
    11→ * Determines if an error is retryable (network/fetch related).
    12→ * Data/logic errors like "not found", "permission denied", or validation
    13→ * errors are NOT retryable — retrying won't fix them.
    14→ */
    15→function isRetryableError(error: unknown): boolean {
    16→  if (!error) return false;
    17→
    18→  const msg = (error instanceof Error ? error.message : String(error)).toLowerCase();
    19→
    20→  // Network / fetch errors — retryable
    21→  if (
    22→    msg.includes('failed to fetch') ||
    23→    msg.includes('networkerror') ||
    24→    msg.includes('network request failed') ||
    25→    msg.includes('net::err_') ||
    26→    msg.includes('fetch error') ||
    27→    msg.includes('timeout') ||
    28→    msg.includes('aborted') ||
    29→    msg.includes('connection') ||
    30→    msg.includes('econnrefused') ||
    31→    msg.includes('econnreset') ||
    32→    msg.includes('socket hang up') ||
    33→    msg.includes('load failed')
    34→  ) {
    35→    return true;
    36→  }
    37→
    38→  // TypeError in fetch context — usually network-related
    39→  if (error instanceof TypeError && (
    40→    msg.includes('fetch') ||
    41→    msg.includes('network') ||
    42→    msg.includes('load') ||
    43→    msg.includes('failed')
    44→  )) {
    45→    return true;
    46→  }
    47→
    48→  // NOT retryable: table doesn't exist, RLS, validation, business logic
    49→  if (
    50→    msg.includes('could not find') ||
    51→    msg.includes('does not exist') ||
    52→    msg.includes('schema cache') ||
    53→    msg.includes('relation') ||
    54→    msg.includes('policy') ||
    55→    msg.includes('permission') ||
    56→    msg.includes('jwt') ||
    57→    msg.includes('rls') ||
    58→    msg.includes('violates') ||
    59→    msg.includes('duplicate') ||
    60→    msg.includes('unique') ||
    61→    msg.includes('not null') ||
    62→    msg.includes('foreign key') ||
    63→    msg.includes('check constraint')
    64→  ) {
    65→    return false;
    66→  }
    67→
    68→  // For Supabase PostgrestError — check code
    69→  const anyErr = error as Record<string, unknown>;
    70→  const code = String(anyErr?.code || '');
    71→  // Network-related Supabase error codes
    72→  if (code === '08' || code.startsWith('08')) return true; // Connection exception
    73→  if (code === '57' || code.startsWith('57')) return true; // Operator intervention
    74→
    75→  // Default: retry unknown errors — they could be transient network issues
    76→  // (Supabase PostgrestError objects are plain objects, not Error instances,
    77→  // and may have empty messages like {}, so we retry them rather than crash)
    78→  return true;
    79→}
    80→
    81→/**
    82→ * Execute an async function with automatic retry on network errors.
    83→ * Uses exponential backoff: 500ms → 1000ms → 2000ms.
    84→ * Only retries on network/fetch errors — data/logic errors fail immediately.
    85→ *
    86→ * @param fn - The async function to execute
    87→ * @param retries - Maximum number of retries (default: 3)
    88→ * @param delay - Initial delay in ms, doubles on each retry (default: 500)
    89→ * @returns The result of fn(), or null if all retries failed
    90→ */
    91→export async function fetchWithRetry<T>(
    92→  fn: () => Promise<T>,
    93→  retries: number = 3,
    94→  delay: number = 500,
    95→): Promise<T | null> {
    96→  try {
    97→    return await fn();
    98→  } catch (error) {
    99→    // If the error is not retryable, log and return null instead of throwing
   100→    // This prevents app crashes — callers use `return result || []` as fallback
   101→    if (!isRetryableError(error)) {
   102→      console.error('[Supabase] ❌ Non-retryable error (returning null):', error instanceof Error ? error.message : error);
   103→      return null;
   104→    }
   105→
   106→    if (retries <= 0) {
   107→      console.error('[Supabase] ❌ All retries exhausted:', error instanceof Error ? error.message : error);
   108→      return null;
   109→    }
   110→
   111→    const attempt = 3 - retries + 1; // Human-readable attempt number
   112→    console.warn(`[Supabase] ⚠️ Network error (attempt ${attempt}/3), retrying in ${delay}ms...`, error instanceof Error ? error.message : error);
   113→
   114→    await new Promise(res => setTimeout(res, delay));
   115→
   116→    return fetchWithRetry(fn, retries - 1, delay * 2);
   117→  }
   118→}

   335→// ============================================
   336→// Helper: camelCase ↔ snake_case conversion
   337→// ============================================
   338→
   339→function toCamelCaseKey(key: string): string {
   340→  return key.replace(/_([a-z])/g, (_, letter) => letter.toUpperCase());
   341→}
   342→
   343→function toSnakeCaseKey(key: string): string {
   344→  return key.replace(/[A-Z]/g, letter => `_${letter.toLowerCase()}`);
   345→}
   346→
   347→function toCamelCase<T extends Record<string, unknown>>(obj: Record<string, unknown>): T {
   348→  const result: Record<string, unknown> = {};
   349→  for (const [key, value] of Object.entries(obj)) {
   350→    result[toCamelCaseKey(key)] = value;
   351→  }
   352→  return result as T;
   353→}
   354→
   355→function toSnakeCase<T extends Record<string, unknown>>(obj: Record<string, unknown>): T {
   356→  const result: Record<string, unknown> = {};
   357→  for (const [key, value] of Object.entries(obj)) {
   358→    if (value !== undefined) {
   359→      result[toSnakeCaseKey(key)] = value;
   360→    }
   361→  }
   362→  return result as T;
   363→}

   369→function isoToDate(value: string | null | undefined): Date | null | undefined {
   370→  if (value === null) return null;
   371→  if (value === undefined) return undefined;
   372→  return new Date(value);
   373→}
   374→
   375→function dateToIso(value: Date | string | null | undefined): string | null | undefined {
   376→  if (value === null) return null;
   377→  if (value === undefined) return undefined;
   378→  if (value instanceof Date) return value.toISOString();
   379→  return new Date(value).toISOString();
   380→}

   594→// ============================================
   595→// calculateFinalBalance - COPIED VERBATIM
   596→// ============================================
   597→
   598→function calculateFinalBalance(
   599→  amount: number,
   600→  conversionFactor: number,
   601→  conversionMethod: string,
   602→  feesType: string,
   603→  feesAmount: number,
   604→  feesDirection: string,
   605→  transactionType: 'INCOME' | 'EXPENSE' = 'INCOME'
   606→): number {
   607→  if (!conversionFactor || conversionFactor === 0) {
   608→    return 0;
   609→  }
   610→  
   611→  let finalBalance = amount;
   612→  
   613→  if (conversionMethod === 'MULTIPLY') {
   614→    finalBalance = amount * conversionFactor;
   615→  } else {
   616→    finalBalance = amount / conversionFactor;
   617→  }
   618→  
   619→  let feesValue = 0;
   620→  if (feesAmount && feesAmount > 0) {
   621→    switch (feesType) {
   622→      case 'FIXED':
   623→        feesValue = feesAmount;
   624→        break;
   625→      case 'PERCENTAGE':
   626→        feesValue = (finalBalance * feesAmount) / 100;
   627→        break;
   628→      case 'PER_THOUSAND':
   629→        feesValue = (finalBalance * feesAmount) / 1000;
   630→        break;
   631→    }
   632→    
   633→    const sameDirection = transactionType === feesDirection;
   634→    
   635→    if (sameDirection) {
   636→      finalBalance = finalBalance + feesValue;
   637→    } else {
   638→      finalBalance = finalBalance - feesValue;
   639→    }
   640→  }
   641→  
   642→  return finalBalance;
   643→}

   688→/**
   689→ * Check whether Supabase tables exist by trying a simple SELECT
   690→ * 🔸 Also checks if Supabase client is configured (isSupabaseConfigured)
   691→ * 🔸 Distinguishes between "tables missing" vs "network/RLS error"
   692→ * 🔸 Logs detailed info for debugging on different devices
   693→ */
   694→export async function checkTablesExist(): Promise<boolean> {
   695→  // First check if Supabase is even configured
   696→  if (!isSupabaseConfigured) {
   697→    console.error('[Supabase] ❌ Client not configured — missing env vars');
   698→    console.error('[Supabase] 💡 Check: NEXT_PUBLIC_SUPABASE_URL and NEXT_PUBLIC_SUPABASE_ANON_KEY must be set');
   699→    console.error('[Supabase] 💡 On Vercel: add in Settings → Environment Variables');
   700→    tablesExist = false;
   701→    return false;
   702→  }
   703→
   704→  try {
   705→    console.log('[Supabase] 🔍 Checking if tables exist...');
   706→    const result = await fetchWithRetry(async () => {
   707→      const { data, error, status } = await supabase.from('currencies').select('id').limit(1);
   708→      if (error) throw error;
   709→      return { data, status };
   710→    });
   711→
   712→    if (result === null) {
   713→      // All retries failed — likely network issue, assume tables exist and retry later
   714→      console.warn('[Supabase] ⚠️ checkTablesExist: network error after retries, assuming tables exist');
   715→      tablesExist = true;
   716→      return true;
   717→    }
   718→
   719→    // Success — tables exist
   720→    console.log('[Supabase] ✅ Tables exist, currencies count:', result.data?.length ?? 0);
   721→    tablesExist = true;
   722→    return true;
   723→  } catch (err) {
   724→    // fetchWithRetry re-throws non-retryable errors (e.g., table doesn't exist, RLS)
   725→    const msg = (err instanceof Error ? err.message : String(err)).toLowerCase();
   726→    
   727→    // Table doesn't exist
   728→    if (
   729→      msg.includes('could not find') ||
   730→      msg.includes('does not exist') ||
   731→      msg.includes('schema cache') ||
   732→      msg.includes('relation')
   733→    ) {
   734→      console.error('[Supabase] 💡 Tables do NOT exist. Run the migration SQL in Supabase SQL Editor.');
   735→      tablesExist = false;
   736→      return false;
   737→    }
   738→
   739→    // RLS or auth errors — tables exist but access denied
   740→    if (msg.includes('policy') || msg.includes('permission') || msg.includes('jwt') || msg.includes('rls') || msg.includes('new row violates')) {
   741→      console.error('[Supabase] ⚠️ Tables EXIST but RLS blocks access — FIX: Run fix-rls.sql in Supabase SQL Editor');
   742→      tablesExist = true;
   743→      return true;
   744→    }
   745→
   746→    // Other errors — don't block the app
   747→    console.error('[Supabase] ❌ checkTablesExist exception:', err);
   748→    tablesExist = true;
   749→    return true;
   750→  }
   751→}

  1117→/**
  1118→ * Recalculate vault balance from opening_balance + all vault-affecting operations after opening_balance_date
  1119→ * Formula: balance = opening_balance + sum(post-date operations)
  1120→ *
  1121→ * Operations that affect vault:
  1122→ * 1. TRANSACTIONS: Complete, cash, not archived
  1123→ * 2. DEBTS: Cash mode debts
  1124→ * 3. DEBT PAYMENTS: Cash mode payments
  1125→ * 4. CURRENCY EXCHANGES: Not deleted
  1126→ */
  1127→export async function recalculateVaultBalance(currencyId: string): Promise<Vault | null> {
  1128→  await initializeDatabase();
  1129→
  1130→  // Get the vault
  1131→  const { data: vaultRows } = await supabase.from('vaults').select('*').eq('currency_id', currencyId);
  1132→  if (!vaultRows || vaultRows.length === 0) return null;
  1133→  const vault = rowToVault(vaultRows[0]);
  1134→
  1135→  const openingBalance = vault.openingBalance || 0;
  1136→  const openingDate = vault.openingBalanceDate;
  1137→
  1138→  let delta = 0;
  1139→
  1140→  // Build date filter - only operations AFTER opening_balance_date
  1141→  const dateFilter = openingDate ? openingDate.toISOString() : null;
  1142→
  1143→  // 1. TRANSACTIONS: Complete, cash (overflow transactions use currencyId directly)
  1144→  let transactionQuery = supabase.from('transactions').select('*')
  1145→    .eq('is_complete', true)
  1146→    .eq('payment_type', 'CASH');
  1147→
  1148→  if (dateFilter) {
  1149→    transactionQuery = transactionQuery.gt('date', dateFilter);
  1150→  }
  1151→
  1152→  const { data: transactions } = await transactionQuery;
  1153→  if (transactions) {
  1154→    for (const tx of transactions) {
  1155→      const txObj = rowToTransaction(tx);
  1156→      // Determine which vault this transaction affects
  1157→      const vaultCurrencyId = txObj.isOverflowTransaction ? txObj.currencyId : (txObj.baseCurrencyId || txObj.currencyId);
  1158→
  1159→      if (vaultCurrencyId === currencyId) {
  1160→        // INCOME: vault decreases (money goes out)
  1161→        // EXPENSE: vault increases (money comes in)
  1162→        if (txObj.type === 'INCOME') {
  1163→          delta -= txObj.amount;
  1164→        } else {
  1165→          delta += txObj.amount;
  1166→        }
  1167→      }
  1168→    }
  1169→  }
  1170→
  1171→  // 2. DEBTS: Cash mode debts affect vault
  1172→  let debtQuery = supabase.from('debts').select('*')
  1173→    .eq('debt_mode', 'CASH');
  1174→
  1175→  if (dateFilter) {
  1176→    debtQuery = debtQuery.gt('date', dateFilter);
  1177→  }
  1178→
  1179→  const { data: debts } = await debtQuery;
  1180→  if (debts) {
  1181→    for (const d of debts) {
  1182→      const debtObj = rowToDebt(d);
  1183→      if (debtObj.currencyId === currencyId) {
  1184→        // PAYABLE (علينا): vault increases (we receive money)
  1185→        // RECEIVABLE (لنا): vault decreases (we give money out)
  1186→        if (debtObj.debtType === 'PAYABLE') {
  1187→          delta += debtObj.amount;
  1188→        } else {
  1189→          delta -= debtObj.amount;
  1190→        }
  1191→      }
  1192→    }
  1193→  }
  1194→
  1195→  // 3. DEBT PAYMENTS: Cash mode
  1196→  let paymentQuery = supabase.from('debt_payments').select('*')
  1197→    .eq('payment_mode', 'CASH');
  1198→
  1199→  if (dateFilter) {
  1200→    paymentQuery = paymentQuery.gt('date', dateFilter);
  1201→  }
  1202→
  1203→  const { data: payments } = await paymentQuery;
  1204→  if (payments) {
  1205→    for (const p of payments) {
  1206→      const paymentObj = rowToDebtPayment(p);
  1207→      if (paymentObj.currencyId === currencyId) {
  1208→        // Debt payment direction logic:
  1209→        // RECEIVABLE (لنا): vault decreases (money goes out to pay toward our receivable)
  1210→        // PAYABLE (علينا): vault increases (money comes in to pay toward our payable)
  1211→        const direction = paymentObj.paymentDirection || 'RECEIVABLE';
  1212→        if (direction === 'PAYABLE') {
  1213→          delta += paymentObj.amount;
  1214→        } else {
  1215→          delta -= paymentObj.amount;
  1216→        }
  1217→      }
  1218→    }
  1219→  }
  1220→
  1221→  // 4. CURRENCY EXCHANGES: Not deleted
  1222→  let exchangeQuery = supabase.from('currency_exchanges').select('*')
  1223→    .eq('is_deleted', false);
  1224→
  1225→  if (dateFilter) {
  1226→    exchangeQuery = exchangeQuery.gt('date', dateFilter);
  1227→  }
  1228→
  1229→  const { data: exchanges } = await exchangeQuery;
  1230→  if (exchanges) {
  1231→    for (const ex of exchanges) {
  1232→      const exchangeObj = rowToCurrencyExchange(ex);
  1233→      // Outgoing: vault decreases
  1234→      if (exchangeObj.outgoingCurrencyId === currencyId) {
  1235→        delta -= exchangeObj.outgoingAmount;
  1236→      }
  1237→      // Incoming: vault increases
  1238→      if (exchangeObj.incomingCurrencyId === currencyId) {
  1239→        delta += exchangeObj.incomingAmount;
  1240→      }
  1241→    }
  1242→  }
  1243→
  1244→  // 5. PURCHASES: Deduct total_price_usd from USD vault only
  1245→  //    (purchases are always paid in USD cash per the spec)
  1246→  let purchaseQuery = supabase.from('purchases').select('*');
  1247→
  1248→  if (dateFilter) {
  1249→    purchaseQuery = purchaseQuery.gt('date', dateFilter);
  1250→  }
  1251→
  1252→  const { data: purchases } = await purchaseQuery;
  1253→  if (purchases) {
  1254→    // Find the USD currency id once
  1255→    const { data: usdCurrencyRow } = await supabase.from('currencies').select('id').eq('code', 'USD').limit(1);
  1256→    const usdCurrencyId = usdCurrencyRow && usdCurrencyRow.length > 0 ? (usdCurrencyRow[0] as Record<string, unknown>).id as string : null;
  1257→    if (usdCurrencyId && currencyId === usdCurrencyId) {
  1258→      for (const p of purchases) {
  1259→        const purchaseObj = rowToPurchase(p);
  1260→        // Purchase: cash goes OUT → vault decreases
  1261→        delta -= purchaseObj.totalPriceUsd;
  1262→      }
  1263→    }
  1264→  }
  1265→
  1266→  // Compute new balance
  1267→  const newBalance = openingBalance + delta;
  1268→
  1269→  // Update vault
  1270→  const { error } = await supabase.from('vaults').update({
  1271→    balance: newBalance,
  1272→    updated_at: new Date().toISOString(),
  1273→  }).eq('id', vault.id);
  1274→
  1275→  if (error) throw new Error(error.message);
  1276→
  1277→  const { data: updatedRow } = await supabase.from('vaults').select('*').eq('id', vault.id).single();
  1278→  return updatedRow ? rowToVault(updatedRow) : null;
  1279→}
