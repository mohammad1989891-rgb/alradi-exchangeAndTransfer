    32→export function AccountStatementModal() {
    33→  const { 
    34→    isAccountStatementOpen, 
    35→    closeAccountStatement, 
    36→    accounts, 
    37→    currencies, 
    38→    selectedAccountForStatement 
    39→  } = useAppStore();
    40→  
    41→  const { transactions, debts, debtPayments } = useSupabaseData();
    42→  
    43→  // Determine initial selected account using useMemo
    44→  const defaultAccountId = useMemo(() => {
    45→    if (selectedAccountForStatement) {
    46→      return selectedAccountForStatement.id;
    47→    }
    48→    return accounts.length > 0 ? accounts[0].id : '';
    49→  }, [selectedAccountForStatement, accounts]);
    50→  
    51→  const [selectedAccountId, setSelectedAccountId] = useState<string>(defaultAccountId);
    52→
    53→  // 🔸 Sales linked to the selected account (all in USD per spec)
    54→  const [accountSales, setAccountSales] = useState<Sale[]>([]);
    55→  const [isLoadingSales, setIsLoadingSales] = useState(false);
    56→
    57→  // Fetch sales whenever the selected account changes
    58→  useEffect(() => {
    59→    if (!selectedAccountId) {
    60→      return;
    61→    }
    62→    let cancelled = false;
    63→    getSalesByAccount(selectedAccountId)
    64→      .then((sales) => {
    65→        if (cancelled) return;
    66→        setAccountSales(sales);
    67→      })
    68→      .catch((err) => {
    69→        console.error('Error fetching sales for account:', err);
    70→        if (!cancelled) setAccountSales([]);
    71→      })
    72→      .finally(() => {
    73→        if (!cancelled) setIsLoadingSales(false);
    74→      });
    75→    return () => { cancelled = true; };
    76→  }, [selectedAccountId]);
    77→  
    78→  // Date filter state
    79→  const [dateFrom, setDateFrom] = useState('');
    80→  const [dateTo, setDateTo] = useState('');
    81→  const hasDateFilter = dateFrom || dateTo;
    82→  
    83→  // Clear date filter
    84→  const clearDateFilter = () => {
    85→    setDateFrom('');
    86→    setDateTo('');
    87→  };
    88→  
    89→  // Filter data for selected account
    90→  const accountTransactions = useMemo(() => {
    91→    if (!selectedAccountId) return [];

    93→    
    94→    // Apply date filter
    95→    if (hasDateFilter) {
    96→      filtered = filtered.filter(t => {
    97→        const txDate = new Date(t.date).toISOString().split('T')[0];
    98→        const matchesDateFrom = !dateFrom || txDate >= dateFrom;
    99→        const matchesDateTo = !dateTo || txDate <= dateTo;
   100→        return matchesDateFrom && matchesDateTo;
   101→      });
   102→    }
   103→    
   104→    return filtered;
   105→  }, [transactions, selectedAccountId, dateFrom, dateTo, hasDateFilter]);
   106→  
   107→  const accountDebts = useMemo(() => {
   108→    if (!selectedAccountId) return [];
   109→    return debts.filter(d => d.accountId === selectedAccountId);
   110→  }, [debts, selectedAccountId]);
   111→  
   112→  // Group transactions by currency
   113→  const transactionsByCurrency = useMemo(() => {
   114→    const grouped: Record<string, Transaction[]> = {};
   115→    
   116→    for (const tx of accountTransactions) {
   117→      const currencyId = tx.currencyId;
   118→      if (!grouped[currencyId]) {
   119→        grouped[currencyId] = [];
   120→      }
   121→      grouped[currencyId].push(tx);
   122→    }
   123→    
   124→    // Sort each group by date
   125→    for (const currencyId in grouped) {
   126→      grouped[currencyId].sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime());
   127→    }
   128→    
   129→    return grouped;
   130→  }, [accountTransactions]);
   131→  
   132→  // 🔸 Date-filtered sales for this account (all sales are USD per spec)
   133→  const filteredAccountSales = useMemo(() => {
   134→    if (!hasDateFilter) return accountSales;
   135→    return accountSales.filter(s => {
   136→      const sDate = new Date(s.date).toISOString().split('T')[0];
   137→      const matchesDateFrom = !dateFrom || sDate >= dateFrom;
   138→      const matchesDateTo = !dateTo || sDate <= dateTo;
   139→      return matchesDateFrom && matchesDateTo;
   140→    });
   141→  }, [accountSales, dateFrom, dateTo, hasDateFilter]);
   142→
   143→  // 🔸 Unified statement item type: a transaction OR a sale, with runningBalance
   144→  type StatementItem = {
   145→    id: string;
   146→    date: Date;
   147→    type: 'INCOME' | 'EXPENSE';
   148→    amount: number;
   149→    finalBalance: number;
   150→    description?: string | null;
   151→    runningBalance: number;
   152→    isSale: boolean;
   153→    paymentMethod?: 'cash' | 'credit';
   154→    materialName?: string;
   155→    quantity?: number;
   156→    unitName?: string;
   157→  };
   158→
   159→  // Calculate totals per currency — for USD, merge sales into the running balance
   160→  const currencyStats = useMemo(() => {
   161→    const stats: Record<string, {
   162→      currency: Currency | undefined;
   163→      totalIncome: number;
   164→      totalExpense: number;
   165→      netBalance: number;
   166→      items: StatementItem[];
   167→    }> = {};
   168→
   169→    // Find USD currency id (sales are always USD)
   170→    const usdCurrency = currencies.find(c => c.code === 'USD');
   171→
   172→    for (const currencyId in transactionsByCurrency) {
   173→      const currency = currencies.find(c => c.id === currencyId);
   174→      const txs = transactionsByCurrency[currencyId];
   175→
   176→      // Build items: convert transactions to StatementItem
   177→      let items: StatementItem[] = txs.map(tx => ({
   178→        id: tx.id,
   179→        date: new Date(tx.date),
   180→        type: tx.type,
   181→        amount: tx.amount,
   182→        finalBalance: tx.finalBalance,
   183→        description: tx.description,
   184→        runningBalance: 0,
   185→        isSale: false,
   186→      }));
   187→
   188→      // 🔸 If this is the USD currency, merge sales (treat each sale as INCOME)
   189→      if (usdCurrency && currencyId === usdCurrency.id && filteredAccountSales.length > 0) {
   190→        const saleItems: StatementItem[] = filteredAccountSales.map(s => ({
   191→          id: s.id,
   192→          date: new Date(s.date),
   193→          type: 'INCOME' as const, // Sale = customer owes us → لنا
   194→          amount: s.totalPrice,
   195→          finalBalance: s.totalPrice,
   196→          description: s.description || `بيع ${s.materialName}`,
   197→          runningBalance: 0,
   198→          isSale: true,
   199→          paymentMethod: s.paymentMethod,
   200→          materialName: s.materialName,
   201→          quantity: s.quantity,
   202→          unitName: s.unitName,
   203→        }));
   204→        items = items.concat(saleItems);
   205→      }
   206→
   207→      // Sort by date ascending (then by createdAt for stable order — use id as tiebreaker)
   208→      items.sort((a, b) => {
   209→        const diff = a.date.getTime() - b.date.getTime();
   210→        if (diff !== 0) return diff;
   211→        return a.id.localeCompare(b.id);
   212→      });
   213→
   214→      // Compute running balance
   215→      let totalIncome = 0;
   216→      let totalExpense = 0;
   217→      let runningBalance = 0;
   218→      items = items.map(it => {
   219→        if (it.type === 'INCOME') {
   220→          totalIncome += it.finalBalance;
   221→          runningBalance += it.finalBalance;
   222→        } else {
   223→          totalExpense += it.finalBalance;
   224→          runningBalance -= it.finalBalance;
   225→        }
   226→        return { ...it, runningBalance };
   227→      });
   228→
   229→      stats[currencyId] = {
   230→        currency,
   231→        totalIncome,
   232→        totalExpense,
   233→        netBalance: runningBalance,
   234→        items,
   235→      };
   236→    }
   237→
   238→    // 🔸 Edge case: account has sales but NO USD transactions — still show USD section
   239→    if (usdCurrency && !stats[usdCurrency.id] && filteredAccountSales.length > 0) {
   240→      let items: StatementItem[] = filteredAccountSales.map(s => ({
   241→        id: s.id,
   242→        date: new Date(s.date),
   243→        type: 'INCOME' as const,
   244→        amount: s.totalPrice,
   245→        finalBalance: s.totalPrice,
   246→        description: s.description || `بيع ${s.materialName}`,
   247→        runningBalance: 0,
   248→        isSale: true,
   249→        paymentMethod: s.paymentMethod,
   250→        materialName: s.materialName,
   251→        quantity: s.quantity,
   252→        unitName: s.unitName,
   253→      }));
   254→      items.sort((a, b) => a.date.getTime() - b.date.getTime());
   255→      let totalIncome = 0;
   256→      let runningBalance = 0;
   257→      items = items.map(it => {
   258→        totalIncome += it.finalBalance;
   259→        runningBalance += it.finalBalance;
   260→        return { ...it, runningBalance };
   261→      });
   262→      stats[usdCurrency.id] = {
   263→        currency: usdCurrency,
   264→        totalIncome,
   265→        totalExpense: 0,
   266→        netBalance: runningBalance,
   267→        items,
   268→      };
   269→    }
   270→
   271→    return stats;
   272→  }, [transactionsByCurrency, currencies, filteredAccountSales]);
   273→  
   274→  // Group debts by currency
   275→  const debtsByCurrency = useMemo(() => {

   279→      const currencyId = debt.currencyId;
   280→      if (!grouped[currencyId]) {
   281→        grouped[currencyId] = [];
   282→      }
   283→      grouped[currencyId].push(debt);
   284→    }
   285→    
   286→    // Sort each group by date
   287→    for (const currencyId in grouped) {
   288→      grouped[currencyId].sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime());
   289→    }
   290→    
   291→    return grouped;
   292→  }, [accountDebts]);
   293→  
   294→  // Calculate debt totals per currency with payments
   295→  const debtStats = useMemo(() => {
   296→    const stats: Record<string, {
   297→      currency: Currency | undefined;
   298→      totalDebt: number;
   299→      paidDebt: number;
   300→      unpaidDebt: number;
   301→      debts: Debt[];
   302→      // إضافة بيانات الدفعات
   303→      paymentsByDebt: Record<string, DebtPayment[]>;
   304→      // حساب الرصيد المتبقي لكل دين
   305→      remainingByDebt: Record<string, number>;
   306→    }> = {};
   307→
   308→    for (const currencyId in debtsByCurrency) {
   309→      const currency = currencies.find(c => c.id === currencyId);
   310→      const currencyDebts = debtsByCurrency[currencyId];
   311→
   312→      // حساب الدفعات لكل دين
   313→      const paymentsByDebt: Record<string, DebtPayment[]> = {};
   314→      const remainingByDebt: Record<string, number> = {};
   315→
   316→      let totalDebt = 0;
   317→      let totalPaid = 0;
   318→
   319→      for (const debt of currencyDebts) {
   320→        // جلب الدفعات المرتبطة بهذا الدين
   321→        const debtPaymentsList = debtPayments.filter(p => p.debtId === debt.id);
   322→        paymentsByDebt[debt.id] = debtPaymentsList;
   323→
   324→        // حساب إجمالي المدفوع لهذا الدين
   325→        const paidAmount = debtPaymentsList.reduce((sum, p) => sum + p.amount, 0);
   326→        const remaining = Math.max(0, debt.finalBalance - paidAmount);
   327→
   328→        remainingByDebt[debt.id] = remaining;
   329→        totalDebt += debt.finalBalance;
   330→        totalPaid += paidAmount;
   331→      }
   332→
   333→      stats[currencyId] = {
   334→        currency,
   335→        totalDebt,
   336→        paidDebt: totalPaid,
   337→        unpaidDebt: totalDebt - totalPaid,
   338→        debts: currencyDebts,
   339→        paymentsByDebt,
   340→        remainingByDebt,
   341→      };
   342→    }
   343→
   344→    return stats;
   345→  }, [debtsByCurrency, currencies, debtPayments]);

   479→        
   480→        ${Object.entries(currencyStats).map(([currencyId, stat]) => `
