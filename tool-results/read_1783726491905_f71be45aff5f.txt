     7→import {
     8→  Select,
     9→  SelectContent,
    10→  SelectItem,

    13→} from '@/components/ui/select';

    19→  DialogFooter,
    20→  DialogDescription,
    21→} from '@/components/ui/dialog';

    26→import {
    27→  Loader2,

    51→export interface SaleDialogProps {
    52→  open: boolean;

    57→
    58→interface FormState {
    59→  date: string;
    60→  accountId: string;
    61→  materialId: string;
    62→  quantity: string;
    63→  unitId: string;
    64→  unitPrice: string;
    65→  paymentMethod: SalePaymentMethod;
    66→  description: string;
    67→}
    68→
    69→function getTodayISO(): string {
    70→  const d = new Date();
    71→  const year = d.getFullYear();
    72→  const month = String(d.getMonth() + 1).padStart(2, '0');
    73→  const day = String(d.getDate()).padStart(2, '0');
    74→  return `${year}-${month}-${day}`;
    75→}
    76→
    77→function toISODateInput(date: Date | string): string {
    78→  try {
    79→    const d = new Date(date);
    80→    if (isNaN(d.getTime())) return getTodayISO();
    81→    const year = d.getFullYear();
    82→    const month = String(d.getMonth() + 1).padStart(2, '0');
    83→    const day = String(d.getDate()).padStart(2, '0');
    84→    return `${year}-${month}-${day}`;
    85→  } catch {
    86→    return getTodayISO();
    87→  }
    88→}
    89→
    90→function getDefaultFormState(): FormState {
    91→  return {
    92→    date: getTodayISO(),
    93→    accountId: '',
    94→    materialId: '',
    95→    quantity: '',
    96→    unitId: '',
    97→    unitPrice: '',
    98→    // 🔸 Default to 'cash' per spec (cash sales add USD to the vault immediately;
    99→    // credit sales are deferred invoices that don't touch the cash box)
   100→    paymentMethod: 'cash',
   101→    description: '',
   102→  };
   103→}
   104→
   105→export function SaleDialog({
   106→  open,
   107→  onOpenChange,
   108→  editingSale,
   109→  onSuccess,
   110→}: SaleDialogProps) {
   111→  const { toast } = useToast();
   112→  const { accounts } = useAppStore();
   113→
   114→  const [materials, setMaterials] = useState<Material[]>([]);
   115→  const [isLoadingMaterials, setIsLoadingMaterials] = useState(false);
   116→  const [inventory, setInventory] = useState<MaterialInventory | null>(null);
   117→  const [isLoadingInventory, setIsLoadingInventory] = useState(false);
   118→  const [isSaving, setIsSaving] = useState(false);
   119→  const [form, setForm] = useState<FormState>(getDefaultFormState());
   120→
   121→  const isEditMode = !!editingSale;
   122→
   123→  // Load materials list when dialog opens
   124→  const loadMaterials = useCallback(async () => {
   125→    setIsLoadingMaterials(true);
   126→    try {
   127→      const data = await getMaterials();
   128→      setMaterials(data);
   129→    } catch (error) {
   130→      console.error('Error loading materials:', error);
   131→      toast({
   132→        title: 'خطأ',
   133→        description: 'حدث خطأ أثناء تحميل قائمة المواد',
   134→        variant: 'destructive',
   135→      });
   136→      setMaterials([]);
   137→    } finally {
   138→      setIsLoadingMaterials(false);
   139→    }
   140→  }, [toast]);
   141→
   142→  // Load inventory for a given material
   143→  const loadInventory = useCallback(async (materialId: string) => {
   144→    if (!materialId) {
   145→      setInventory(null);
   146→      return;
   147→    }
   148→    setIsLoadingInventory(true);
   149→    try {
   150→      const inv = await getMaterialInventory(materialId);
   151→      setInventory(inv);
   152→    } catch (error) {
   153→      console.error('Error loading inventory:', error);
   154→      setInventory(null);
   155→    } finally {
   156→      setIsLoadingInventory(false);
   157→    }
   158→  }, []);
   159→
   160→  // Initialize form when dialog opens
   161→  useEffect(() => {
   162→    if (!open) return;
   163→    loadMaterials();
   164→    if (editingSale) {
   165→      setForm({
   166→        date: toISODateInput(editingSale.date),
   167→        accountId: editingSale.accountId,
   168→        materialId: editingSale.materialId,
   169→        quantity: String(editingSale.quantity ?? ''),
   170→        unitId: editingSale.unitId,
   171→        unitPrice: String(editingSale.unitPrice ?? ''),
   172→        paymentMethod: editingSale.paymentMethod === 'credit' ? 'credit' : 'cash',
   173→        description: editingSale.description ?? '',
   174→      });
   175→      // Preload inventory for the editing sale's material
   176→      loadInventory(editingSale.materialId);
   177→    } else {
   178→      setForm(getDefaultFormState());
   179→      setInventory(null);
   180→    }
   181→  }, [open, editingSale, loadMaterials, loadInventory]);
   182→
   183→  // Find the currently selected material
   184→  const selectedMaterial = useMemo<Material | null>(() => {
   185→    if (!form.materialId) return null;
   186→    return materials.find((m) => m.id === form.materialId) || null;
   187→  }, [form.materialId, materials]);
   188→
   189→  // Find the currently selected material-unit (for baseFactor lookup)
   190→  const selectedMaterialUnit = useMemo(() => {
   191→    if (!selectedMaterial || !form.unitId) return null;
   192→    return (
   193→      selectedMaterial.materialUnits?.find((mu) => mu.unitId === form.unitId) ||
   194→      null
   195→    );
   196→  }, [selectedMaterial, form.unitId]);
   197→
   198→  // When material changes, reset unitId to the material's default unit and re-fetch inventory
   199→  const handleMaterialChange = (materialId: string) => {
   200→    const mat = materials.find((m) => m.id === materialId) || null;
   201→    let defaultUnitId = '';
   202→    if (mat) {
   203→      if (mat.defaultUnitId) {
   204→        defaultUnitId = mat.defaultUnitId;
   205→      } else {
   206→        const baseUnit = mat.materialUnits?.find((mu) => mu.baseFactor === 1);
   207→        defaultUnitId = baseUnit?.unitId || mat.materialUnits?.[0]?.unitId || '';
   208→      }
   209→    }
   210→    setForm((prev) => ({
   211→      ...prev,
   212→      materialId,
   213→      unitId: defaultUnitId,
   214→    }));
   215→    loadInventory(materialId);
   216→  };
   217→
   218→  // Live numeric parsing
   219→  const quantityNum = useMemo(() => {
   220→    const n = parseFloat(form.quantity);
   221→    return isNaN(n) ? 0 : n;
   222→  }, [form.quantity]);
   223→
   224→  const unitPriceNum = useMemo(() => {
   225→    const n = parseFloat(form.unitPrice);
   226→    return isNaN(n) ? 0 : n;
   227→  }, [form.unitPrice]);
   228→
   229→  // Live total price calculation
   230→  const totalPrice = useMemo(() => {
   231→    return quantityNum * unitPriceNum;
   232→  }, [quantityNum, unitPriceNum]);
   233→
   234→  // Inventory check: convert entered quantity to base using selected unit's baseFactor
   235→  const quantityInBase = useMemo(() => {
   236→    const factor = selectedMaterialUnit?.baseFactor ?? 0;
   237→    return quantityNum * factor;
   238→  }, [quantityNum, selectedMaterialUnit]);
   239→
   240→  const exceedsInventory = useMemo(() => {
   241→    if (!inventory || !form.materialId || quantityNum <= 0) return false;
   242→    // Allow small floating-point tolerance
   243→    return quantityInBase > inventory.currentInBase + 0.0001;
   244→  }, [inventory, form.materialId, quantityNum, quantityInBase]);
   245→
   246→  // 🔸 Stock in the material's conversion unit (e.g. برميل) — purely
   247→  //    informational. Uses the shared computeConversionUnitStock() helper so
   248→  //    the SAME logic + formatting is used across SaleDialog, PurchaseDialog,
   249→  //    and the inventory summary cards on PurchasesPage & SalesPage.
   250→  //    Picks the material's first unit whose baseFactor > 1 (i.e. NOT the
   251→  //    default unit). Returns null when no conversion unit exists → the
   252→  //    second field shows "غير معرف" (per spec: "إذا لم تكن المادة تحتوي
   253→  //    على وحدة تحويل ... يظهر بقيمة غير معرف").
   254→  const stockInSelectedUnit = useMemo<{ value: string | null; unitName: string }>(() => {
   255→    if (!inventory) return { value: null, unitName: '' };
   256→    const result = computeConversionUnitStock(
   257→      inventory.currentInBase,
   258→      inventory.material.materialUnits,
   259→      inventory.material.defaultUnitId,
   260→    );
   261→    return {
   262→      value: result?.value ?? null,
   263→      unitName: result?.unitName ?? '',
   264→    };
   265→  }, [inventory]);
   266→
   267→  const handleFieldChange = (field: keyof FormState, value: string) => {
   268→    setForm((prev) => ({ ...prev, [field]: value }));
   269→  };
   270→
   271→  const validate = (): string | null => {
   272→    if (!form.date) return 'يرجى اختيار التاريخ';
   273→    if (!form.accountId) return 'يرجى اختيار الحساب';
   274→    if (!form.materialId) return 'يرجى اختيار المادة';
   275→    if (!form.unitId) return 'يرجى اختيار الواحدة';
   276→    if (!form.quantity || isNaN(quantityNum) || quantityNum <= 0) {
   277→      return 'يرجى إدخال الكمية (أكبر من صفر)';
   278→    }
   279→    if (form.unitPrice === '' || isNaN(unitPriceNum) || unitPriceNum < 0) {
   280→      return 'يرجى إدخال سعر إفرادي صحيح';
   281→    }
   282→    if (exceedsInventory) {
   283→      return 'الكمية تتجاوز المخزون المتوفر';
   284→    }
   285→    return null;
   286→  };
   287→
   288→  const handleSave = async () => {
   289→    const validationError = validate();
   290→    if (validationError) {
   291→      toast({
   292→        title: 'بيانات ناقصة',
   293→        description: validationError,
   294→        variant: 'destructive',
   295→      });
   296→      return;
   297→    }
   298→
   299→    setIsSaving(true);
   300→    try {
   301→      const payload = {
   302→        date: form.date,
   303→        accountId: form.accountId,
   304→        materialId: form.materialId,
   305→        quantity: quantityNum,
   306→        unitId: form.unitId,
   307→        unitPrice: unitPriceNum,
   308→        paymentMethod: form.paymentMethod,
   309→        description: form.description.trim() || undefined,
   310→      };
   311→
   312→      if (isEditMode && editingSale) {
   313→        await updateSale(editingSale.id, payload);
   314→        toast({
   315→          title: 'تم التحديث',
   316→          description: 'تم تحديث عملية البيع بنجاح',
   317→        });
   318→      } else {
   319→        await addSale(payload);
   320→        toast({
   321→          title: 'تم الحفظ',
   322→          description: 'تم إضافة عملية البيع بنجاح',
   323→        });
   324→      }
   325→
   326→      // Dispatch refresh events so other components can update
   327→      window.dispatchEvent(new Event('sales-updated'));
   328→      window.dispatchEvent(new Event('app-data-refreshed'));
   329→
   330→      onSuccess?.();
   331→      onOpenChange(false);
   332→    } catch (error) {
   333→      console.error('Error saving sale:', error);
   334→      toast({
   335→        title: 'خطأ',
   336→        description:
   337→          error instanceof Error
   338→            ? error.message
   339→            : 'حدث خطأ أثناء حفظ عملية البيع',
   340→        variant: 'destructive',
   341→      });
   342→    } finally {
   343→      setIsSaving(false);
   344→    }
   345→  };
   346→
   347→  // Only show active accounts in the dropdown (defensive; spec doesn't require filtering)
   348→  const activeAccounts = useMemo(() => accounts, [accounts]);
   349→
   350→  return (
   351→    <Dialog open={open} onOpenChange={onOpenChange}>
   352→      <DialogContent scrollable className="max-w-md">
   353→        {/* 🔸 Pinned header — stays visible while body scrolls */}
   354→        <DialogHeader className="flex-shrink-0 border-b px-6 py-4 text-right">
   355→          <DialogTitle className="flex items-center gap-2">
   356→            <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-emerald-500 to-green-600 flex items-center justify-center">
   357→              <TrendingUp className="w-4 h-4 text-white" />
   358→            </div>
   359→            {isEditMode ? 'تعديل عملية بيع' : 'إضافة عملية بيع'}
   360→          </DialogTitle>
   361→          <DialogDescription className="sr-only">
   362→            {isEditMode
   363→              ? 'نموذج تعديل عملية بيع'
   364→              : 'نموذج إضافة عملية بيع جديدة'}
   365→          </DialogDescription>
   366→        </DialogHeader>
   367→
   368→        {/* 🔸 Scrollable body — vertical scroll when content overflows */}
   369→        <div className="min-h-0 flex-1 overflow-y-auto overscroll-contain px-6 py-4 [scrollbar-width:thin]">
   370→          {isLoadingMaterials ? (
   371→            <div className="flex items-center justify-center py-12">
   372→              <Loader2 className="w-6 h-6 animate-spin text-muted-foreground" />
   373→            </div>
   374→          ) : (
   375→            <div className="space-y-4">
   376→              {/* التاريخ */}
   377→              <div className="space-y-1.5">
   378→                <Label htmlFor="sale-date" className="flex items-center gap-1.5 text-sm">
   379→                  <Calendar className="w-3.5 h-3.5 text-muted-foreground" />
   380→                  التاريخ <span className="text-red-500">*</span>
   381→                </Label>
   382→                <Input
   383→                  id="sale-date"
   384→                  type="date"
   385→                  value={form.date}
   386→                  onChange={(e) => handleFieldChange('date', e.target.value)}
   387→                  className="rounded-xl"
   388→                />
   389→              </div>
   390→
   391→              {/* اسم الحساب */}
   392→              <div className="space-y-1.5">
   393→                <Label className="flex items-center gap-1.5 text-sm">
   394→                  <User className="w-3.5 h-3.5 text-muted-foreground" />
   395→                  اسم الحساب <span className="text-red-500">*</span>
   396→                </Label>
   397→                <Select
   398→                  value={form.accountId}
   399→                  onValueChange={(v) => handleFieldChange('accountId', v)}
   400→                >
   401→                  <SelectTrigger className="rounded-xl">
   402→                    <SelectValue placeholder="اختر الحساب" />
   403→                  </SelectTrigger>
   404→                  <SelectContent>
   405→                    {activeAccounts.length === 0 ? (
   406→                      <div className="px-3 py-2 text-sm text-muted-foreground">
   407→                        لا توجد حسابات مسجلة
   408→                      </div>
   409→                    ) : (
   410→                      activeAccounts.map((a) => (
   411→                        <SelectItem key={a.id} value={a.id}>
   412→                          {a.name}
   413→                        </SelectItem>
   414→                      ))
   415→                    )}
   416→                  </SelectContent>
   417→                </Select>
   418→              </div>
   419→
   420→              {/* اسم المادة */}
   421→              <div className="space-y-1.5">
   422→                <Label className="flex items-center gap-1.5 text-sm">
   423→                  <Package className="w-3.5 h-3.5 text-muted-foreground" />
   424→                  اسم المادة <span className="text-red-500">*</span>
   425→                </Label>
   426→                <Select
   427→                  value={form.materialId}
   428→                  onValueChange={handleMaterialChange}
   429→                >
   430→                  <SelectTrigger className="rounded-xl">
   431→                    <SelectValue placeholder="اختر المادة" />
   432→                  </SelectTrigger>
   433→                  <SelectContent>
   434→                    {materials.length === 0 ? (
   435→                      <div className="px-3 py-2 text-sm text-muted-foreground">
   436→                        لا توجد مواد مسجلة
   437→                      </div>
   438→                    ) : (
   439→                      materials.map((m) => (
   440→                        <SelectItem key={m.id} value={m.id}>
   441→                          {m.name}
   442→                        </SelectItem>
   443→                      ))
   444→                    )}
   445→                  </SelectContent>
   446→                </Select>
   447→              </div>
   448→
   449→              {/* Inventory info box — default unit + conversion unit side by side */}
   450→              {form.materialId && (
   451→                isLoadingInventory ? (
   452→                  <div className="flex items-center gap-2 rounded-xl border border-muted-foreground/20 bg-muted/30 text-muted-foreground px-3 py-2 text-xs">
   453→                    <Loader2 className="w-3.5 h-3.5 animate-spin" />
   454→                    <span>جاري تحميل المخزون...</span>
   455→                  </div>
   456→                ) : inventory ? (
   457→                  <div className="grid grid-cols-2 gap-2">
   458→                    {/* 🔸 Existing field: stock in default unit (content unchanged) */}
   459→                    <div className="flex items-center gap-1.5 rounded-xl border border-emerald-200/70 bg-emerald-50/70 text-emerald-700 dark:border-emerald-800/40 dark:bg-emerald-950/20 dark:text-emerald-300 px-2.5 py-2 text-xs min-w-0">
   460→                      <Info className="w-3.5 h-3.5 flex-shrink-0" />
   461→                      <span className="min-w-0 truncate">
   462→                        المتوفر: {formatNumber(inventory.currentInDefaultUnit)}{' '}
   463→                        <span className="font-medium">{inventory.defaultUnitName}</span>
   464→                      </span>
   465→                    </div>
   466→                    {/* 🔸 New field: stock in the selected conversion unit */}
   467→                    <div className="flex items-center gap-1.5 rounded-xl border border-emerald-200/70 bg-emerald-50/70 text-emerald-700 dark:border-emerald-800/40 dark:bg-emerald-950/20 dark:text-emerald-300 px-2.5 py-2 text-xs min-w-0">
   468→                      <Info className="w-3.5 h-3.5 flex-shrink-0" />
   469→                      {stockInSelectedUnit.value !== null ? (
   470→                        <span className="min-w-0 truncate">
   471→                          بوحدة التحويل: {stockInSelectedUnit.value}{' '}
   472→                          <span className="font-medium">{stockInSelectedUnit.unitName}</span>
   473→                        </span>
   474→                      ) : (
   475→                        <span className="min-w-0 truncate text-muted-foreground">
   476→                          بوحدة التحويل: غير معرف
   477→                        </span>
   478→                      )}
   479→                    </div>
   480→                  </div>
   481→                ) : (
   482→                  <div className="flex items-center gap-2 rounded-xl border border-emerald-200/70 bg-emerald-50/70 text-emerald-700 dark:border-emerald-800/40 dark:bg-emerald-950/20 dark:text-emerald-300 px-3 py-2 text-xs">
   483→                    <Info className="w-3.5 h-3.5" />
   484→                    <span>لا توجد بيانات مخزون</span>
   485→                  </div>
   486→                )
   487→              )}
   488→
   489→              {/* الكمية + الواحدة */}
   490→              <div className="grid grid-cols-2 gap-3">
   491→                <div className="space-y-1.5">
   492→                  <Label
   493→                    htmlFor="sale-quantity"
   494→                    className="flex items-center gap-1.5 text-sm"
   495→                  >
   496→                    <Boxes className="w-3.5 h-3.5 text-muted-foreground" />
   497→                    الكمية <span className="text-red-500">*</span>
   498→                  </Label>
   499→                  <Input
   500→                    id="sale-quantity"
   501→                    type="number"
   502→                    min="0"
   503→                    step="any"
   504→                    value={form.quantity}
   505→                    onChange={(e) => handleFieldChange('quantity', e.target.value)}
   506→                    placeholder="0"
   507→                    className={cn(
   508→                      'rounded-xl',
   509→                      exceedsInventory &&
   510→                        'border-red-500 focus-visible:ring-red-500/40'
   511→                    )}
   512→                  />
   513→                  {exceedsInventory && (
   514→                    <p className="flex items-center gap-1 text-xs text-red-600 dark:text-red-400 font-medium">
   515→                      <AlertTriangle className="w-3 h-3" />
   516→                      الكمية تتجاوز المخزون المتوفر
   517→                    </p>
   518→                  )}
   519→                </div>
   520→                <div className="space-y-1.5">
   521→                  <Label className="flex items-center gap-1.5 text-sm">
   522→                    <Tag className="w-3.5 h-3.5 text-muted-foreground" />
   523→                    الواحدة <span className="text-red-500">*</span>
   524→                  </Label>
   525→                  <Select
   526→                    value={form.unitId}
   527→                    onValueChange={(v) => handleFieldChange('unitId', v)}
   528→                    disabled={!selectedMaterial}
   529→                  >
   530→                    <SelectTrigger className="rounded-xl">
   531→                      <SelectValue placeholder="اختر الواحدة" />
   532→                    </SelectTrigger>
   533→                    <SelectContent>
   534→                      {selectedMaterial?.materialUnits &&
   535→                      selectedMaterial.materialUnits.length > 0 ? (
   536→                        selectedMaterial.materialUnits.map((mu) => (
   537→                          <SelectItem key={mu.id} value={mu.unitId}>
   538→                            {mu.unit?.name || 'وحدة'}
   539→                          </SelectItem>
   540→                        ))
   541→                      ) : (
   542→                        <div className="px-3 py-2 text-sm text-muted-foreground">
   543→                          لا توجد وحدات لهذه المادة
   544→                        </div>
   545→                      )}
   546→                    </SelectContent>
   547→                  </Select>
   548→                </div>
   549→              </div>
   550→
   551→              {/* السعر الإفرادي */}
   552→              <div className="space-y-1.5">
