    32→export function AccountStatementModal() {
    33→  const { 
    34→    isAccountStatementOpen, 
    35→    closeAccountStatement, 
    36→    accounts, 
    37→    currencies, 
    38→    selectedAccountForStatement

    40→  
    41→  const { transactions, debts, debtPayments } = useSupabaseData();
    42→  
    43→  // Determine initial selected account using useMemo
    44→  const defaultAccountId = useMemo(() => {
    45→    if (selectedAccountForStatement) {
    46→      return selectedAccountForStatement.id;
    47→    }
    48→    return accounts.length > 0 ? accounts[0].id : '';
    49→  }, [selectedAccountForStatement, accounts]);
    50→  
    51→  const [selectedAccountId, setSelectedAccountId] = useState<string>(defaultAccountId);
    52→
    53→  // 🔸 Sales linked to the selected account (all in USD per spec)
    54→  const [accountSales, setAccountSales] = useState<Sale[]>([]);
    55→  const [isLoadingSales, setIsLoadingSales] = useState(false);
    56→
    57→  // 🔸 Keep the statement's sales section in sync with the latest data.
    58→  //    The modal fetches sales from the `sales` table via getSalesByAccount
    59→  //    (which selects ALL sales for the account — no payment_method filter —
    60→  //    so both cash and credit sales are always returned).
    61→  //
    62→  //    Root cause of "credit sales don't appear in the statement": the
    63→  //    original effect only refetched when `selectedAccountId` changed, so
    64→  //    sales created/edited/deleted while the modal was closed (or while a
    65→  //    different account was selected) never appeared until the user
    66→  //    manually switched accounts. The fix below adds two more triggers:
    67→  //      1. Refetch when the modal opens (isAccountStatementOpen turns true)
    68→  //         — picks up sales created since the last view.
    69→  //      2. Listen for `sales-updated` + `app-data-refreshed` window events
    70→  //         (dispatched by SaleDialog and SalesPage after create/edit/delete)
    71→  //         and refetch live, so the statement + final balance update
    72→  //         immediately without requiring the user to reopen the modal.
    73→  useEffect(() => {
    74→    if (!selectedAccountId) {
    75→      // No account selected — nothing to fetch. (We intentionally do NOT call
    76→      // setAccountSales([]) here to avoid the set-state-in-effect lint rule;
    77→      // accountSales is reset via the empty result path of getSalesByAccount
    78→      // when a real account is selected, and the modal is hidden when no
    79→      // account exists anyway.)
    80→      return;
    81→    }
    82→    let cancelled = false;
    83→
    84→    const load = () => {
    85→      setIsLoadingSales(true);
    86→      getSalesByAccount(selectedAccountId)
    87→        .then((sales) => {
    88→          if (!cancelled) setAccountSales(sales);
    89→        })
    90→        .catch((err) => {
    91→          console.error('Error fetching sales for account:', err);
    92→          if (!cancelled) setAccountSales([]);
    93→        })
    94→        .finally(() => {
    95→          if (!cancelled) setIsLoadingSales(false);
    96→        });
    97→    };
    98→
    99→    // Initial load (fires on mount, on account change, and on modal open)
   100→    load();
   101→
   102→    // Live refresh: refetch whenever a sale is created/edited/deleted anywhere
   103→    // in the app, so the statement's balance stays accurate in real time.
   104→    const handleSalesUpdated = () => load();
   105→    window.addEventListener('sales-updated', handleSalesUpdated);
   106→    window.addEventListener('app-data-refreshed', handleSalesUpdated);
   107→
   108→    return () => {
   109→      cancelled = true;
   110→      window.removeEventListener('sales-updated', handleSalesUpdated);
   111→      window.removeEventListener('app-data-refreshed', handleSalesUpdated);
   112→    };
   113→  }, [selectedAccountId, isAccountStatementOpen]);
   114→  
   115→  // Date filter state
   116→  const [dateFrom, setDateFrom] = useState('');
   117→  const [dateTo, setDateTo] = useState('');
   118→  const hasDateFilter = dateFrom || dateTo;
   119→

   121→  const clearDateFilter = () => {

   123→    setDateTo('');

   125→  
   126→  // Filter data for selected account
   127→  const accountTransactions = useMemo(() => {
   128→    if (!selectedAccountId) return [];
   129→    let filtered = transactions.filter(t => t.accountId === selectedAccountId && t.isComplete !== false);
   130→    
   131→    // Apply date filter
   132→    if (hasDateFilter) {
   133→      filtered = filtered.filter(t => {
   134→        const txDate = new Date(t.date).toISOString().split('T')[0];
   135→        const matchesDateFrom = !dateFrom || txDate >= dateFrom;
   136→        const matchesDateTo = !dateTo || txDate <= dateTo;
   137→        return matchesDateFrom && matchesDateTo;
   138→      });
   139→    }
   140→    
   141→    return filtered;
   142→  }, [transactions, selectedAccountId, dateFrom, dateTo, hasDateFilter]);
   143→  
   144→  const accountDebts = useMemo(() => {
   145→    if (!selectedAccountId) return [];
   146→    return debts.filter(d => d.accountId === selectedAccountId);
   147→  }, [debts, selectedAccountId]);
   148→  
   149→  // Group transactions by currency
   150→  const transactionsByCurrency = useMemo(() => {
   151→    const grouped: Record<string, Transaction[]> = {};
   152→    
   153→    for (const tx of accountTransactions) {
   154→      const currencyId = tx.currencyId;
   155→      if (!grouped[currencyId]) {
   156→        grouped[currencyId] = [];
   157→      }
   158→      grouped[currencyId].push(tx);
   159→    }
   160→    
   161→    // Sort each group by date
   162→    for (const currencyId in grouped) {
   163→      grouped[currencyId].sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime());
   164→    }
   165→    
   166→    return grouped;
   167→  }, [accountTransactions]);
   168→  
   169→  // 🔸 Date-filtered sales for this account (all sales are USD per spec)
   170→  const filteredAccountSales = useMemo(() => {
   171→    if (!hasDateFilter) return accountSales;
   172→    return accountSales.filter(s => {
   173→      const sDate = new Date(s.date).toISOString().split('T')[0];
   174→      const matchesDateFrom = !dateFrom || sDate >= dateFrom;
   175→      const matchesDateTo = !dateTo || sDate <= dateTo;
   176→      return matchesDateFrom && matchesDateTo;
   177→    });
   178→  }, [accountSales, dateFrom, dateTo, hasDateFilter]);
   179→
   180→  // 🔸 Unified statement item type: a transaction OR a sale, with runningBalance
   181→  type StatementItem = {
   182→    id: string;
   183→    date: Date;
   184→    type: 'INCOME' | 'EXPENSE';
   185→    amount: number;
   186→    finalBalance: number;
   187→    description?: string | null;
   188→    runningBalance: number;
   189→    isSale: boolean;
   190→    paymentMethod?: 'cash' | 'credit';
   191→    materialName?: string;
   192→    quantity?: number;
   193→    unitName?: string;
   194→  };
   195→
   196→  // Calculate totals per currency — for USD, merge sales into the running balance
   197→  const currencyStats = useMemo(() => {
   198→    const stats: Record<string, {
   199→      currency: Currency | undefined;
   200→      totalIncome: number;
   201→      totalExpense: number;
   202→      netBalance: number;
   203→      items: StatementItem[];
   204→    }> = {};
   205→
   206→    // Find USD currency id (sales are always USD)
   207→    const usdCurrency = currencies.find(c => c.code === 'USD');
   208→
   209→    for (const currencyId in transactionsByCurrency) {
   210→      const currency = currencies.find(c => c.id === currencyId);
   211→      const txs = transactionsByCurrency[currencyId];
   212→
   213→      // Build items: convert transactions to StatementItem
   214→      let items: StatementItem[] = txs.map(tx => ({
   215→        id: tx.id,
   216→        date: new Date(tx.date),
   217→        type: tx.type,
   218→        amount: tx.amount,
   219→        finalBalance: tx.finalBalance,
   220→        description: tx.description,
   221→        runningBalance: 0,
   222→        isSale: false,
   223→      }));
   224→
   225→      // 🔸 If this is the USD currency, merge sales (treat each sale as INCOME)
   226→      if (usdCurrency && currencyId === usdCurrency.id && filteredAccountSales.length > 0) {
   227→        const saleItems: StatementItem[] = filteredAccountSales.map(s => ({
   228→          id: s.id,
   229→          date: new Date(s.date),
   230→          type: 'INCOME' as const, // Sale = customer owes us → لنا
   231→          amount: s.totalPrice,
   232→          finalBalance: s.totalPrice,
   233→          description: s.description || `بيع ${s.materialName}`,
   234→          runningBalance: 0,
   235→          isSale: true,
   236→          paymentMethod: s.paymentMethod,
   237→          materialName: s.materialName,
   238→          quantity: s.quantity,
   239→          unitName: s.unitName,
   240→        }));
   241→        items = items.concat(saleItems);
   242→      }
   243→
   244→      // Sort by date ascending (then by createdAt for stable order — use id as tiebreaker)
   245→      items.sort((a, b) => {
   246→        const diff = a.date.getTime() - b.date.getTime();
   247→        if (diff !== 0) return diff;
   248→        return a.id.localeCompare(b.id);
   249→      });
   250→
   251→      // Compute running balance
   252→      // 🔸 Cash sales are reference-only: they remain visible in the statement
   253→      //    as a historical record (with all invoice data + a "كاش" badge) but
   254→      //    do NOT affect totalIncome / runningBalance, because their value was
   255→      //    already collected directly into the USD vault at sale time.
   256→      //    Only credit sales (unpaid receivables) move the account balance,
   257→      //    since كشف الحساب exists to show الذمم والمبالغ المستحقة.
   258→      let totalIncome = 0;
   259→      let totalExpense = 0;
   260→      let runningBalance = 0;
   261→      items = items.map(it => {
   262→        const isCashSale = it.isSale && it.paymentMethod === 'cash';
   263→        if (!isCashSale) {
   264→          if (it.type === 'INCOME') {
   265→            totalIncome += it.finalBalance;
   266→            runningBalance += it.finalBalance;
   267→          } else {
   268→            totalExpense += it.finalBalance;
   269→            runningBalance -= it.finalBalance;
   270→          }
   271→        }
   272→        return { ...it, runningBalance };
   273→      });
   274→
   275→      stats[currencyId] = {
   276→        currency,
   277→        totalIncome,
   278→        totalExpense,
   279→        netBalance: runningBalance,
   280→        items,
   281→      };
   282→    }
   283→
   284→    // 🔸 Edge case: account has sales but NO USD transactions — still show USD section
   285→    if (usdCurrency && !stats[usdCurrency.id] && filteredAccountSales.length > 0) {
   286→      let items: StatementItem[] = filteredAccountSales.map(s => ({
   287→        id: s.id,
   288→        date: new Date(s.date),
   289→        type: 'INCOME' as const,
   290→        amount: s.totalPrice,
   291→        finalBalance: s.totalPrice,
   292→        description: s.description || `بيع ${s.materialName}`,
   293→        runningBalance: 0,
   294→        isSale: true,
   295→        paymentMethod: s.paymentMethod,
   296→        materialName: s.materialName,
   297→        quantity: s.quantity,
   298→        unitName: s.unitName,
   299→      }));
   300→      items.sort((a, b) => a.date.getTime() - b.date.getTime());
   301→      // 🔸 Cash sales are reference-only (see main loop comment above) —
   302→      //    visible in the statement but excluded from the balance.
   303→      let totalIncome = 0;
   304→      let runningBalance = 0;
   305→      items = items.map(it => {
   306→        const isCashSale = it.isSale && it.paymentMethod === 'cash';
   307→        if (!isCashSale) {
   308→          totalIncome += it.finalBalance;
   309→          runningBalance += it.finalBalance;
   310→        }
   311→        return { ...it, runningBalance };
   312→      });
   313→      stats[usdCurrency.id] = {
   314→        currency: usdCurrency,
   315→        totalIncome,
   316→        totalExpense: 0,
   317→        netBalance: runningBalance,
   318→        items,
   319→      };
   320→    }
   321→
   322→    return stats;
   323→  }, [transactionsByCurrency, currencies, filteredAccountSales]);
   324→  
   325→  // Group debts by currency
   326→  const debtsByCurrency = useMemo(() => {
   327→    const grouped: Record<string, Debt[]> = {};
   328→    
   329→    for (const debt of accountDebts) {
   330→      const currencyId = debt.currencyId;
   331→      if (!grouped[currencyId]) {
   332→        grouped[currencyId] = [];
   333→      }
   334→      grouped[currencyId].push(debt);
   335→    }
   336→    
   337→    // Sort each group by date
   338→    for (const currencyId in grouped) {
   339→      grouped[currencyId].sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime());
   340→    }
   341→    
   342→    return grouped;
   343→  }, [accountDebts]);
   344→  
   345→  // Calculate debt totals per currency with payments
   346→  const debtStats = useMemo(() => {
   347→    const stats: Record<string, {
   348→      currency: Currency | undefined;
   349→      totalDebt: number;
   350→      paidDebt: number;
   351→      unpaidDebt: number;
   352→      debts: Debt[];
   353→      // إضافة بيانات الدفعات
   354→      paymentsByDebt: Record<string, DebtPayment[]>;
   355→      // حساب الرصيد المتبقي لكل دين
   356→      remainingByDebt: Record<string, number>;
   357→    }> = {};
   358→
   359→    for (const currencyId in debtsByCurrency) {
   360→      const currency = currencies.find(c => c.id === currencyId);
   361→      const currencyDebts = debtsByCurrency[currencyId];
   362→
   363→      // حساب الدفعات لكل دين
   364→      const paymentsByDebt: Record<string, DebtPayment[]> = {};
   365→      const remainingByDebt: Record<string, number> = {};
   366→
   367→      let totalDebt = 0;
   368→      let totalPaid = 0;
   369→
   370→      for (const debt of currencyDebts) {
   371→        // جلب الدفعات المرتبطة بهذا الدين
   372→        const debtPaymentsList = debtPayments.filter(p => p.debtId === debt.id);
   373→        paymentsByDebt[debt.id] = debtPaymentsList;
   374→
   375→        // حساب إجمالي المدفوع لهذا الدين
   376→        const paidAmount = debtPaymentsList.reduce((sum, p) => sum + p.amount, 0);
   377→        const remaining = Math.max(0, debt.finalBalance - paidAmount);
   378→
   379→        remainingByDebt[debt.id] = remaining;
   380→        totalDebt += debt.finalBalance;
   381→        totalPaid += paidAmount;
   382→      }
   383→
   384→      stats[currencyId] = {
   385→        currency,
   386→        totalDebt,
   387→        paidDebt: totalPaid,
   388→        unpaidDebt: totalDebt - totalPaid,
   389→        debts: currencyDebts,
   390→        paymentsByDebt,
   391→        remainingByDebt,
   392→      };
   393→    }
   394→
   395→    return stats;
   396→  }, [debtsByCurrency, currencies, debtPayments]);

   398→  const selectedAccount = accounts.find(a => a.id === selectedAccountId);
