    55→    55→function toEnglishNumbers(num: number | string): string {
    56→    56→  const arabicNumerals = ['٠', '١', '٢', '٣', '٤', '٥', '٦', '٧', '٨', '٩'];
    57→    57→  let str = typeof num === 'number' ? num.toString() : num;
    58→    58→  
    59→    59→  // تحويل الأرقام العربية إلى إنجليزية
    60→    60→  for (let i = 0; i < 10; i++) {
    61→    61→    str = str.replace(new RegExp(arabicNumerals[i], 'g'), i.toString());
    62→    62→  }
    63→    63→  
    64→    64→  // تنسيق الرقم مع فواصل
    65→    65→  if (typeof num === 'number') {
    66→    66→    return num.toLocaleString('en-US');
    67→    67→  }
    68→    68→  return str;
    69→    69→}
